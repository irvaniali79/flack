    24→    24→  try {
    25→    25→    const raw = localStorage.getItem(DRAFT_KEY)
    26→    26→    const parsed = raw ? JSON.parse(raw) : {}
    27→    27→    return typeof parsed === 'object' && parsed ? parsed : {}

   159→   159→function replaceMessage(list: MessageDTO[], messageId: string, next: MessageDTO): MessageDTO[] {
   160→   160→  const idx = list.findIndex((m) => m.id === messageId)
   161→   161→  if (idx === -1) return list
   162→   162→  const copy = [...list]
   163→   163→  copy[idx] = next
   164→   164→  return copy
   165→   165→}
   166→   166→
   167→   167→function bumpReplyCount(list: MessageDTO[], rootId: string, delta: number): MessageDTO[] {
   168→   168→  const idx = list.findIndex((m) => m.id === rootId)
   169→   169→  if (idx === -1) return list
   170→   170→  const copy = [...list]
   171→   171→  copy[idx] = { ...copy[idx], replyCount: Math.max(0, copy[idx].replyCount + delta) }
   172→   172→  return copy
   173→   173→}

   175→   175→function touchChannel(channels: ChannelDTO[], channelId: string, message: MessageDTO): ChannelDTO[] {
   176→   176→  return channels.map((c) =>
   177→   177→    c.id === channelId
   178→   178→      ? {
   179→   179→          ...c,
   180→   180→          lastMessage: {
   181→   181→            body: message.body,
   182→   182→            createdAt: message.createdAt,
   183→   183→            senderName: message.sender?.name ?? null,
   184→   184→          },
   185→   185→        }
   186→   186→      : c,
   187→   187→  )
   188→   188→}

   323→   323→      divider = messages[idx]?.id ?? null
   324→   324→    }

   328→   328→      unreadDividerByChannel: { ...s.unreadDividerByChannel, [channelId]: divider },

   380→   380→  sendMessage: async (body, fileIds, parentId) => {
   381→   381→    const channelId = get().activeChannelId
   382→   382→    if (!channelId) return
   383→   383→    const data = await api<{ message: MessageDTO }>(`/api/channels/${channelId}/messages`, {
   384→   384→      method: 'POST',
   385→   385→      body: { body, fileIds, parentId },
   386→   386→    })
   387→   387→    const message = data.message
   388→   388→    set((s) => {
   389→   389→      if (message.parentId) {
   390→   390→        const replies = s.threadReplies[message.parentId] ?? []
   391→   391→        if (replies.some((r) => r.id === message.id)) return {}
   392→   392→        return {
   393→   393→          threadReplies: { ...s.threadReplies, [message.parentId]: [...replies, message] },
   394→   394→          messagesByChannel: {
   395→   395→            ...s.messagesByChannel,
   396→   396→            [channelId]: bumpReplyCount(s.messagesByChannel[channelId] ?? [], message.parentId, 1),
   397→   397→          },
   398→   398→          channels: touchChannel(s.channels, channelId, message),
   399→   399→        }
   400→   400→      }
   401→   401→      const list = s.messagesByChannel[channelId] ?? []
   402→   402→      if (list.some((m) => m.id === message.id)) return {}
   403→   403→      return {
   404→   404→        messagesByChannel: { ...s.messagesByChannel, [channelId]: [...list, message] },
   405→   405→        channels: touchChannel(s.channels, channelId, message),
   406→   406→      }
   407→   407→    })
   408→   408→    // Clear typing indicator for me
   409→   409→    get().handleTyping(channelId, message.sender?.id ?? 'me', '', undefined, true)
   410→   410→  },

   421→   421→  deleteMessage: async (messageId) => {
   422→   422→    await api(`/api/messages/${messageId}`, { method: 'DELETE' })
   423→   423→    const { messagesByChannel } = get()
   424→   424→    const entry = Object.entries(messagesByChannel).find(([, list]) =>
   425→   425→      list.some((m) => m.id === messageId),
   426→   426→    )
   427→   427→    const channelId = entry?.[0]
   428→   428→    if (channelId) get().handleMessageDeleted(messageId, channelId)
   429→   429→    // also drop from thread replies if present
   430→   430→    set((s) => {
   431→   431→      const nextThreadReplies: Record<string, MessageDTO[]> = {}
   432→   432→      for (const [rootId, replies] of Object.entries(s.threadReplies)) {
   433→   433→        nextThreadReplies[rootId] = replies.map((r) =>
   434→   434→          r.id === messageId ? { ...r, deletedAt: new Date().toISOString(), isPinned: false } : r,
   435→   435→        )
   436→   436→      }
   437→   437→      return { threadReplies: nextThreadReplies }
   438→   438→    })
   439→   439→  },
   440→   440→
   441→   441→  toggleReaction: async (messageId, emoji) => {
   442→   442→    const data = await api<{ reactions: ReactionDTO[] }>(`/api/messages/${messageId}/reactions`, {
   443→   443→      method: 'POST',
   444→   444→      body: { emoji },
   445→   445→    })
   446→   446→    const { messagesByChannel, threadReplies } = get()
   447→   447→    for (const [channelId] of Object.entries(messagesByChannel)) {
   448→   448→      if (messagesByChannel[channelId]?.some((m) => m.id === messageId)) {
   449→   449→        get().handleReactionUpdated(channelId, messageId, data.reactions)
   450→   450→      }
   451→   451→    }
   452→   452→    for (const [rootId, replies] of Object.entries(threadReplies)) {
   453→   453→      if (replies.some((m) => m.id === messageId)) {
   454→   454→        set((s) => ({
   455→   455→          threadReplies: {
   456→   456→            ...s.threadReplies,
   457→   457→            [rootId]: replies.map((r) => (r.id === messageId ? { ...r, reactions: data.reactions } : r)),
   458→   458→          },
   459→   459→        }))
   460→   460→      }
   461→   461→    }
   462→   462→  },
   463→   463→

   632→   632→            },
   633→   633→          }
   634→   634→        }

   638→   638→          threadReplies: { ...s.threadReplies, [message.parentId]: [...replies, message] },
   639→   639→          messagesByChannel: {

   647→   647→        }

   662→   662→              ),
   663→   663→        }
   664→   664→      }

   666→   666→      const isActive = s.activeChannelId === message.channelId
   667→   667→      const shouldCountUnread = !isMine && !isActive
   668→   668→      return {
   669→   669→        channels: shouldCountUnread
   670→   670→          ? channels.map((c) =>
   671→   671→              c.id === message.channelId && c.isMember ? { ...c, unread: c.unread + 1 } : c,
   672→   672→            )
   673→   673→          : channels,
   674→   674→        messagesByChannel: { ...s.messagesByChannel, [message.channelId]: [...list, message] },
   675→   675→      }
   676→   676→    })

   682→   682→  },

   766→   766→export function channelDisplayName(channel: ChannelDTO | null, meId: string | null): string {
   767→   767→  if (!channel) return ''
   768→   768→  if (channel.kind === 'dm') {
   769→   769→    const others = channel.members ?? []
   770→   770→    if (others.length === 1) return others[0].name
   771→   771→    return 'Direct message'
   772→   772→  }
   773→   773→  if (channel.kind === 'group_dm') {
   774→   774→    const others = (channel.members ?? []).map((m) => m.name.split(' ')[0])
   775→   775→    return others.join(', ') || channel.name
   776→   776→  }
   777→   777→  return channel.name
   778→   778→}

     1→     1→// ─── Connector catalog (Slack-style App Directory) ───────────────────────────
     2→     2→// A connector is an external service (Google Calendar, GitHub, …) installed
     3→     3→// into the workspace. Each install ("connection") links an account label to a
     4→     4→// destination channel and posts its events there as a bot "app" user
     5→     5→// (User.kind = 'app'), exactly like Slack app messages.
     6→     6→//
     7→     7→// This module is server-safe (no React). Icons are referenced by key — the
     8→     8→// client maps them to lucide components in the Connectors view.
     9→     9→import { db } from '@/lib/db'
    10→    10→import { emitToChannel } from '@/lib/realtime-server'

    14→    14→// ─── Types ───────────────────────────────────────────────────────────────────
    15→    15→
    16→    16→export interface ConnectorEventDef {
    17→    17→  id: string
    18→    18→  label: string
    19→    19→  description: string
    20→    20→  defaultOn: boolean
    21→    21→}
    22→    22→
    23→    23→export interface ConnectorDef {
    24→    24→  id: string
    25→    25→  name: string
    26→    26→  tagline: string
    27→    27→  description: string
    28→    28→  category: string
    29→    29→  /** icon key resolved by the client icon map */
    30→    30→  icon: string
    31→    31→  /** brand hex used for the app tile */
    32→    32→  color: string
    33→    33→  /** demo OAuth scope labels shown in the connect flow */
    34→    34→  scopes: string[]
    35→    35→  /** realistic demo account labels for the connect flow */
    36→    36→  sampleAccounts: string[]
    37→    37→  events: ConnectorEventDef[]
    38→    38→}
    39→    39→
    40→    40→/** A concrete event instance — the content of one app message */
    41→    41→export interface ConnectorEventInstance {
    42→    42→  eventId: string
    43→    43→  title: string
    44→    44→  /** plain-text version (searchable, notifications, sidebar previews) */
    45→    45→  body: string
    46→    46→  fields?: { label: string; value: string }[]
    47→    47→  actions?: { label: string; style: 'primary' | 'default' }[]
    48→    48→  footer?: string
    49→    49→}

   964→   964→export function parseEventSubs(raw: string | null): Record<string, boolean> {
   965→   965→  if (!raw) return {}
   966→   966→  try {
   967→   967→    const parsed = JSON.parse(raw) as Record<string, unknown>
   968→   968→    const out: Record<string, boolean> = {}
   969→   969→    for (const [k, v] of Object.entries(parsed)) out[k] = v === true
   970→   970→    return out
   971→   971→  } catch {
   972→   972→    return {}
   973→   973→  }
   974→   974→}

    23→function loadDrafts(): Record<string, string> {
    24→  try {
    25→    const raw = localStorage.getItem(DRAFT_KEY)
    26→    const parsed = raw ? JSON.parse(raw) : {}
    27→    return typeof parsed === 'object' && parsed ? parsed : {}
    28→  } catch {
    29→    return {}
    30→  }
    31→}
    32→
    33→function saveDrafts(drafts: Record<string, string>) {
    34→  try {
    35→    localStorage.setItem(DRAFT_KEY, JSON.stringify(drafts))
    36→  } catch {
    37→    // ignore
    38→  }
    39→}

   159→function replaceMessage(list: MessageDTO[], messageId: string, next: MessageDTO): MessageDTO[] {
   160→  const idx = list.findIndex((m) => m.id === messageId)
   161→  if (idx === -1) return list
   162→  const copy = [...list]
   163→  copy[idx] = next
   164→  return copy
   165→}
   166→
   167→function bumpReplyCount(list: MessageDTO[], rootId: string, delta: number): MessageDTO[] {
   168→  const idx = list.findIndex((m) => m.id === rootId)
   169→  if (idx === -1) return list
   170→  const copy = [...list]
   171→  copy[idx] = { ...copy[idx], replyCount: Math.max(0, copy[idx].replyCount + delta) }
   172→  return copy
   173→}

   175→function touchChannel(channels: ChannelDTO[], channelId: string, message: MessageDTO): ChannelDTO[] {
   176→  return channels.map((c) =>
   177→    c.id === channelId
   178→      ? {
   179→          ...c,
   180→          lastMessage: {
   181→            body: message.body,
   182→            createdAt: message.createdAt,
   183→            senderName: message.sender?.name ?? null,
   184→          },
   185→        }
   186→      : c,
   187→  )
   188→}

   256→
   257→  logout: async () => {
   258→    try {
   259→      await api('/api/auth/logout', { method: 'POST' })

   262→    }
   263→    disconnectSocket()
   264→    set({
   265→      me: null,
   266→      users: [],
   267→      channels: [],
   268→      agents: [],
   269→      activeChannelId: null,
   270→      activeThreadRootId: null,
   271→      messagesByChannel: {},
   272→      threadReplies: {},
   273→      presence: {},
   274→      typing: {},
   275→      notifications: [],

   284→  fetchChannels: async () => {
   285→    const { me } = get()
   286→    if (!me) return
   287→    try {
   288→      const data = await api<{ channels: ChannelDTO[] }>('/api/channels')
   289→      set({ channels: data.channels })

   291→      // ignore — keep stale list
   292→    }
   293→  },
   294→
   295→  fetchNotifications: async () => {
   296→    try {
   297→      const data = await api<{ notifications: NotificationDTO[] }>('/api/notifications')
   298→      set({
   299→        notifications: data.notifications,
   300→        notificationsUnread: data.notifications.filter((n) => !n.readAt).length,

   302→    } catch {
   303→      // ignore
   304→    }

   306→
   307→  // ── channels ───────────────────────────────────────────────────────────────
   308→  openChannel: async (channelId) => {
   309→    const prev = get().activeChannelId
   310→    if (prev && prev !== channelId) socketLeaveChannel(prev)
   311→    set({ activeChannelId: channelId, activeThreadRootId: null, drawerOpen: false, editingMessageId: null })
   312→    socketJoinChannel(channelId)
   313→
   314→    const channel = get().channels.find((c) => c.id === channelId)
   315→    const unread = channel?.unread ?? 0
   316→    await get().fetchMessages(channelId)
   317→
   318→    // Track the unread divider position before marking read
   319→    const messages = get().messagesByChannel[channelId] ?? []
   320→    let divider: string | null = null
   321→    if (unread > 0 && messages.length > 0) {
   322→      const idx = Math.max(0, messages.length - unread)
   323→      divider = messages[idx]?.id ?? null
   324→    }
   325→    const last = messages[messages.length - 1]
   326→    if (last) get().markChannelRead(channelId, last.id)
   327→    set((s) => ({
   328→      unreadDividerByChannel: { ...s.unreadDividerByChannel, [channelId]: divider },
   329→    }))
   330→  },
   331→
   332→  closeChannel: () => {
   333→    const prev = get().activeChannelId
   334→    if (prev) socketLeaveChannel(prev)
   335→    set({ activeChannelId: null, activeThreadRootId: null })
   336→  },
   337→
   338→  fetchMessages: async (channelId) => {
   339→    if (get().messagesLoading[channelId]) return
   340→    set((s) => ({ messagesLoading: { ...s.messagesLoading, [channelId]: true } }))
   341→    try {
   342→      const data = await api<{ messages: MessageDTO[]; hasMore: boolean }>(
   343→        `/api/channels/${channelId}/messages?limit=50`,
   344→      )
   345→      set((s) => ({
   346→        messagesByChannel: { ...s.messagesByChannel, [channelId]: data.messages },
   347→        hasMoreByChannel: { ...s.hasMoreByChannel, [channelId]: data.hasMore },
   348→        messagesLoading: { ...s.messagesLoading, [channelId]: false },
   349→      }))
   350→    } catch {
   351→      set((s) => ({ messagesLoading: { ...s.messagesLoading, [channelId]: false } }))
   352→    }
   353→  },
   354→
   355→  loadOlder: async (channelId) => {
   356→    const { messagesByChannel, hasMoreByChannel, loadingOlder } = get()
   357→    if (!hasMoreByChannel[channelId] || loadingOlder[channelId]) return
   358→    const messages = messagesByChannel[channelId] ?? []
   359→    const oldest = messages[0]
   360→    if (!oldest) return
   361→    set((s) => ({ loadingOlder: { ...s.loadingOlder, [channelId]: true } }))
   362→    try {
   363→      const data = await api<{ messages: MessageDTO[]; hasMore: boolean }>(
   364→        `/api/channels/${channelId}/messages?limit=50&beforeId=${oldest.id}`,
   365→      )
   366→      set((s) => ({
   367→        messagesByChannel: {
   368→          ...s.messagesByChannel,
   369→          [channelId]: [...data.messages, ...(s.messagesByChannel[channelId] ?? [])],
   370→        },
   371→        hasMoreByChannel: { ...s.hasMoreByChannel, [channelId]: data.hasMore },
   372→        loadingOlder: { ...s.loadingOlder, [channelId]: false },
   373→      }))
   374→    } catch {
   375→      set((s) => ({ loadingOlder: { ...s.loadingOlder, [channelId]: false } }))
   376→    }
   377→  },
   378→
   379→  // ── messages ───────────────────────────────────────────────────────────────
   380→  sendMessage: async (body, fileIds, parentId) => {
   381→    const channelId = get().activeChannelId
   382→    if (!channelId) return
   383→    const data = await api<{ message: MessageDTO }>(`/api/channels/${channelId}/messages`, {
   384→      method: 'POST',
   385→      body: { body, fileIds, parentId },
   386→    })
   387→    const message = data.message
   388→    set((s) => {
   389→      if (message.parentId) {
   390→        const replies = s.threadReplies[message.parentId] ?? []
   391→        if (replies.some((r) => r.id === message.id)) return {}
   392→        return {
   393→          threadReplies: { ...s.threadReplies, [message.parentId]: [...replies, message] },
   394→          messagesByChannel: {
   395→            ...s.messagesByChannel,
   396→            [channelId]: bumpReplyCount(s.messagesByChannel[channelId] ?? [], message.parentId, 1),
   397→          },
   398→          channels: touchChannel(s.channels, channelId, message),
   399→        }
   400→      }
   401→      const list = s.messagesByChannel[channelId] ?? []
   402→      if (list.some((m) => m.id === message.id)) return {}
   403→      return {
   404→        messagesByChannel: { ...s.messagesByChannel, [channelId]: [...list, message] },
   405→        channels: touchChannel(s.channels, channelId, message),
   406→      }
   407→    })
   408→    // Clear typing indicator for me
   409→    get().handleTyping(channelId, message.sender?.id ?? 'me', '', undefined, true)
   410→  },
   411→
   412→  editMessage: async (messageId, body) => {
   413→    const data = await api<{ message: MessageDTO }>(`/api/messages/${messageId}`, {
   414→      method: 'PATCH',
   415→      body: { body },
   416→    })

   418→    set({ editingMessageId: null })
   419→  },
   420→
   421→  deleteMessage: async (messageId) => {
   422→    await api(`/api/messages/${messageId}`, { method: 'DELETE' })
   423→    const { messagesByChannel } = get()
   424→    const entry = Object.entries(messagesByChannel).find(([, list]) =>
   425→      list.some((m) => m.id === messageId),
   426→    )
   427→    const channelId = entry?.[0]
   428→    if (channelId) get().handleMessageDeleted(messageId, channelId)
   429→    // also drop from thread replies if present
   430→    set((s) => {
   431→      const nextThreadReplies: Record<string, MessageDTO[]> = {}
   432→      for (const [rootId, replies] of Object.entries(s.threadReplies)) {
   433→        nextThreadReplies[rootId] = replies.map((r) =>
   434→          r.id === messageId ? { ...r, deletedAt: new Date().toISOString(), isPinned: false } : r,
   435→        )
   436→      }
   437→      return { threadReplies: nextThreadReplies }
   438→    })
   439→  },
   440→
   441→  toggleReaction: async (messageId, emoji) => {
   442→    const data = await api<{ reactions: ReactionDTO[] }>(`/api/messages/${messageId}/reactions`, {
   443→      method: 'POST',
   444→      body: { emoji },
   445→    })
   446→    const { messagesByChannel, threadReplies } = get()
   447→    for (const [channelId] of Object.entries(messagesByChannel)) {
   448→      if (messagesByChannel[channelId]?.some((m) => m.id === messageId)) {
   449→        get().handleReactionUpdated(channelId, messageId, data.reactions)
   450→      }
   451→    }
   452→    for (const [rootId, replies] of Object.entries(threadReplies)) {
   453→      if (replies.some((m) => m.id === messageId)) {
   454→        set((s) => ({
   455→          threadReplies: {
   456→            ...s.threadReplies,
   457→            [rootId]: replies.map((r) => (r.id === messageId ? { ...r, reactions: data.reactions } : r)),
   458→          },
   459→        }))
   460→      }
   461→    }
   462→  },
   463→
   464→  togglePin: async (messageId, pinned) => {
   465→    const data = await api<{ message: MessageDTO }>(`/api/messages/${messageId}/pin`, {
   466→      method: 'POST',
   467→      body: { pinned },
   468→    })
   469→    get().handleMessageUpdated(data.message)
   470→  },

   473→  openThread: async (rootId) => {

   477→      set((s) => ({ threadReplies: { ...s.threadReplies, [rootId]: data.replies }, threadLoading: false }))
   478→    } catch {
   479→      set({ threadLoading: false })
   480→    }
   481→  },
   482→
   483→  closeThread: () => set({ activeThreadRootId: null, editingMessageId: null }),

   486→  createChannel: async (name, topic, kind, memberIds) => {
   487→    try {
   488→      const data = await api<{ id: string }>('/api/channels', {
   489→        method: 'POST',
   490→        body: { name, topic, kind, memberIds },

   493→      await get().openChannel(data.id)

   497→    }

   501→    const data = await api<{ channel: ChannelDTO }>('/api/dms', { method: 'POST', body: { userIds } })
   502→    const channel = data.channel
   503→    set((s) => ({
   504→      channels: s.channels.some((c) => c.id === channel.id) ? s.channels : [...s.channels, channel],
   505→    }))
   506→    await get().openChannel(channel.id)
   507→    return channel.id

   510→  joinChannel: async (channelId) => {
   511→    await api(`/api/channels/${channelId}/join`, { method: 'POST' })
   512→    await get().fetchChannels()
   513→    await get().openChannel(channelId)
   514→  },

   516→  leaveChannel: async (channelId) => {
   517→    await api(`/api/channels/${channelId}/leave`, { method: 'POST' })
   518→    const wasActive = get().activeChannelId === channelId
   519→    await get().fetchChannels()
   520→    if (wasActive) {
   521→      const general = get().channels.find((c) => c.slug === 'general')
   522→      if (general) await get().openChannel(general.id)
   523→      else set({ activeChannelId: null })
   524→    }
   525→  },

   527→  addChannelMember: async (channelId, userId) => {

   529→    await get().fetchChannels()
   530→  },

   532→  updateChannel: async (channelId, patch) => {
   533→    const data = await api<{ name: string; topic: string | null; isArchived: boolean }>(
   534→      `/api/channels/${channelId}`,
   535→      { method: 'PATCH', body: patch },
   536→    )
   537→    set((s) => ({
   538→      channels: s.channels.map((c) =>
   539→        c.id === channelId
   540→          ? { ...c, name: data.name, topic: data.topic, isArchived: data.isArchived }
   541→          : c,
   542→      ),
   543→    }))
   544→  },
   545→
   546→  setNotifyPrefs: async (channelId, patch) => {
   547→    // optimistic
   548→    set((s) => ({
   549→      channels: s.channels.map((c) => (c.id === channelId ? { ...c, ...patch } : c)),
   550→    }))
   551→    try {
   552→      await api(`/api/channels/${channelId}/members`, { method: 'PATCH', body: patch })
   553→    } catch {
   554→      await get().fetchChannels()
   555→    }
   556→  },
   557→
   558→  markChannelRead: (channelId, messageId) => {
   559→    set((s) => ({
   560→      channels: s.channels.map((c) =>
   561→        c.id === channelId ? { ...c, unread: 0, mentionCount: 0 } : c,
   562→      ),
   563→    }))
   564→    if (messageId) {

   614→  handleNewMessage: (message) => {
   615→    const me = get().me
   616→    set((s) => {
   617→      const isMine = !!me && message.sender?.id === me.id
   618→      // Thread reply
   619→      if (message.parentId) {
   620→        const replies = s.threadReplies[message.parentId]
   621→        const channels = touchChannel(s.channels, message.channelId, message)
   622→        if (!replies) {
   623→          return {
   624→            channels,
   625→            messagesByChannel: {
   626→              ...s.messagesByChannel,
   627→              [message.channelId]: bumpReplyCount(
   628→                s.messagesByChannel[message.channelId] ?? [],
   629→                message.parentId,
   630→                1,
   631→              ),
   632→            },
   633→          }
   634→        }
   635→        if (replies.some((r) => r.id === message.id)) return {}
   636→        return {
   637→          channels,
   638→          threadReplies: { ...s.threadReplies, [message.parentId]: [...replies, message] },
   639→          messagesByChannel: {
   640→            ...s.messagesByChannel,
   641→            [message.channelId]: bumpReplyCount(
   642→              s.messagesByChannel[message.channelId] ?? [],
   643→              message.parentId,
   644→              1,
   645→            ),
   646→          },
   647→        }
   648→      }
   649→
   650→      // Top-level message
   651→      const list = s.messagesByChannel[message.channelId]
   652→      const channels = touchChannel(s.channels, message.channelId, message)
   653→      if (!list) {
   654→        // Channel not loaded — bump unread counters instead
   655→        return {
   656→          channels: isMine
   657→            ? channels
   658→            : channels.map((c) =>
   659→                c.id === message.channelId && c.isMember
   660→                  ? { ...c, unread: c.unread + 1 }
   661→                  : c,
   662→              ),
   663→        }
   664→      }
   665→      if (list.some((m) => m.id === message.id)) return {}
   666→      const isActive = s.activeChannelId === message.channelId
   667→      const shouldCountUnread = !isMine && !isActive
   668→      return {
   669→        channels: shouldCountUnread
   670→          ? channels.map((c) =>
   671→              c.id === message.channelId && c.isMember ? { ...c, unread: c.unread + 1 } : c,
   672→            )
   673→          : channels,
   674→        messagesByChannel: { ...s.messagesByChannel, [message.channelId]: [...list, message] },
   675→      }
   676→    })
   677→    // Clear the sender's typing indicator
   678→    if (message.sender) {
   679→      get().handleTyping(message.channelId, message.sender.id, '', undefined, true)
   680→    }
   681→    // If channel is active but window hidden, mark unread for badge (skip — keep simple)
   682→  },

   766→export function channelDisplayName(channel: ChannelDTO | null, meId: string | null): string {
   767→  if (!channel) return ''
   768→  if (channel.kind === 'dm') {
   769→    const others = channel.members ?? []
   770→    if (others.length === 1) return others[0].name
   771→    return 'Direct message'
   772→  }
   773→  if (channel.kind === 'group_dm') {
   774→    const others = (channel.members ?? []).map((m) => m.name.split(' ')[0])
   775→    return others.join(', ') || channel.name
   776→  }
   777→  return channel.name
   778→}

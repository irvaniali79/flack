    30→// ─── typing indicator ────────────────────────────────────────────────────────
    31→
    32→export function TypingIndicator({ channelId }: { channelId: string }) {
    33→  const typing = useChatStore((s) => s.typing[channelId])
    34→  const me = useChatStore((s) => s.me)
    35→  const [, setTick] = useState(0)
    36→
    37→  useEffect(() => {
    38→    const interval = setInterval(() => setTick((v) => v + 1), 1000)
    39→    return () => clearInterval(interval)
    40→  }, [])
    41→
    42→  const entries = Object.entries(typing ?? {})
    43→    .filter(([userId, entry]) => userId !== me?.id && Date.now() - entry.at < 4000)
    44→    .map(([, entry]) => entry)
    45→
    46→  if (entries.length === 0) return null
    47→  const agent = entries.find((entry) => entry.kind === 'agent')
    48→
    49→  return (
    50→    <div className="flex items-center gap-2 px-4 pb-1 text-xs text-muted-foreground md:px-5" aria-live="polite">
    51→      {agent ? (
    52→        <Sparkles className="h-3.5 w-3.5 text-amber-500" aria-hidden />
    53→      ) : (
    54→        <span className="flex gap-0.5" aria-hidden>
    55→          <span className="typing-dot h-1 w-1 rounded-full bg-emerald-500" />
    56→          <span className="typing-dot h-1 w-1 rounded-full bg-emerald-500" />
    57→          <span className="typing-dot h-1 w-1 rounded-full bg-emerald-500" />
    58→        </span>
    59→      )}
    60→      {entries.length === 1
    61→        ? `${entries[0].name} is ${agent ? 'thinking…' : 'typing…'}`
    62→        : agent
    63→          ? `${agent.name} is thinking…`
    64→          : `${entries.length} people are typing…`}
    65→    </div>
    66→  )
    67→}

    77→function MentionMenu({
    78→  users,
    79→  query,
    80→  onSelect,
    81→  onClose,
    82→  activeIndex,
    83→}: {
    84→  users: UserDTO[]
    85→  query: string
    86→  onSelect: (user: UserDTO | 'channel' | 'here') => void
    87→  onClose: () => void
    88→  activeIndex: number
    89→}) {
    90→  const q = query.toLowerCase()
    91→  const filtered = users
    92→    .filter((user) => !q || user.name.toLowerCase().includes(q) || (user.handle ?? '').toLowerCase().includes(q))
    93→    .slice(0, 7)
    94→
    95→  return (
    96→    <div className="absolute bottom-full left-0 z-30 mb-2 w-64 overflow-hidden rounded-xl border border-border bg-popover shadow-xl">
    97→      <p className="border-b border-border px-3 py-1.5 text-[10px] font-bold uppercase tracking-wider text-muted-foreground">
    98→        Mention
    99→      </p>
   100→      <div className="max-h-60 overflow-y-auto p-1">
   101→        {q.length >= 2 && (
   102→          <>
   103→            <MenuItem
   104→              label="@channel"
   105→              description="Notify everyone in this channel"
   106→              special
   107→              active={activeIndex === 0}
   108→              onSelect={() => onSelect('channel')}
   109→            />
   110→            <MenuItem
   111→              label="@here"
   112→              description="Notify people active now"
   113→              special
   114→              active={activeIndex === 1}
   115→              onSelect={() => onSelect('here')}
   116→            />
   117→          </>
   118→        )}
   119→        {filtered.map((user, index) => (
   120→          <button
   121→            key={user.id}
   122→            type="button"
   123→            className={cn(
   124→              'flex w-full items-center gap-2.5 rounded-lg px-2 py-1.5 text-left transition-colors duration-150',
   125→              activeIndex === index + (q.length >= 2 ? 2 : 0) ? 'bg-accent' : 'hover:bg-accent',
   126→            )}
   127→            onMouseDown={(event) => event.preventDefault()}
   128→            onClick={() => onSelect(user)}
   129→          >
   130→            <UserAvatar user={user} size="xs" />
   131→            <span className="min-w-0 flex-1">
   132→              <span className="block truncate text-sm">{user.name}</span>
   133→              {user.handle && (
   134→                <span className="block truncate text-[11px] text-muted-foreground">@{user.handle}</span>
   135→              )}
   136→            </span>
   137→            {user.kind === 'agent' && (
   138→              <span className="flex items-center gap-0.5 rounded bg-amber-500/15 px-1 py-px text-[9px] font-bold uppercase text-amber-600 dark:text-amber-400">
   139→                <Sparkles className="h-2.5 w-2.5" aria-hidden /> AI
   140→              </span>
   141→            )}
   142→          </button>
   143→        ))}
   144→        {filtered.length === 0 && q.length >= 2 && null}
   145→        {filtered.length === 0 && (
   146→          <p className="px-3 py-3 text-center text-xs text-muted-foreground">No matches</p>
   147→        )}
   148→      </div>
   149→      <button
   150→        type="button"
   151→        className="w-full border-t border-border px-3 py-1.5 text-left text-[11px] text-muted-foreground hover:bg-accent"
   152→        onMouseDown={(event) => event.preventDefault()}
   153→        onClick={onClose}
   154→      >
   155→        Esc to dismiss
   156→      </button>
   157→    </div>
   158→  )
   159→}

   201→// ─── composer ────────────────────────────────────────────────────────────────
   202→
   203→export function Composer({ parentId, placeholder }: { parentId?: string; placeholder?: string }) {
   204→  const channel = useChatStore(getActiveChannel)
   205→  const channelId = channel?.id ?? ''
   206→  const me = useChatStore((s) => s.me)
   207→  const users = useChatStore((s) => s.users)
   208→  const channels = useChatStore((s) => s.channels)
   209→  const sendMessage = useChatStore((s) => s.sendMessage)
   210→  const editMessage = useChatStore((s) => s.editMessage)
   211→  const emitTyping = useChatStore((s) => s.emitTyping)
   212→  const drafts = useChatStore((s) => s.drafts)
   213→  const setDraft = useChatStore((s) => s.setDraft)
   214→  const editingMessageId = useChatStore((s) => s.editingMessageId)
   215→  const setEditingMessageId = useChatStore((s) => s.setEditingMessageId)
   216→  const messages = useChatStore((s) => (channelId ? s.messagesByChannel[channelId] : undefined)) ?? []
   217→  const threadReplies = useChatStore((s) => (s.activeThreadRootId ? s.threadReplies[s.activeThreadRootId] : undefined))
   218→
   219→  const draftKey = parentId ? `${channelId}-thread-${parentId}` : channelId
   220→  const [text, setText] = useState('')
   221→  const [files, setFiles] = useState<UploadedFile[]>([])
   222→  const [uploading, setUploading] = useState(0)
   223→  const [menu, setMenu] = useState<MenuState | null>(null)
   224→  const [menuIndex, setMenuIndex] = useState(0)
   225→  const [dragging, setDragging] = useState(false)
   226→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   227→  const fileInputRef = useRef<HTMLInputElement>(null)
   228→  const lastTypingEmit = useRef(0)
   229→
   230→  // editing state: load message into the box
   231→  const editingMessage = useMemo(() => {
   232→    if (!editingMessageId) return null
   233→    const inChannel = messages.find((m) => m.id === editingMessageId)
   234→    const inThread = threadReplies?.find((m) => m.id === editingMessageId)
   235→    return inChannel ?? inThread ?? null
   236→  }, [editingMessageId, messages, threadReplies])
   237→
   238→  useEffect(() => {
   239→    if (editingMessage) {
   240→      setText(editingMessage.body)
   241→      textareaRef.current?.focus()
   242→    }
   243→  }, [editingMessage])
   244→
   245→  // restore draft when channel changes (skip when editing)
   246→  useEffect(() => {
   247→    if (!editingMessage) {
   248→      setText(drafts[draftKey] ?? '')
   249→    }
   250→    setFiles([])
   251→    setMenu(null)
   252→    // eslint-disable-next-line react-hooks/exhaustive-deps
   253→  }, [draftKey])
   254→
   255→  const isAgentDm = channel?.kind === 'dm' && channel.members?.some((m) => m.kind === 'agent')
   256→
   257→  const autosize = () => {
   258→    const el = textareaRef.current
   259→    if (!el) return
   260→    el.style.height = 'auto'
   261→    el.style.height = `${Math.min(el.scrollHeight, 160)}px`
   262→  }
   263→
   264→  useEffect(autosize, [text])
   265→
   266→  const detectMenu = (value: string, caret: number) => {
   267→    const before = value.slice(0, caret)
   268→    const mentionMatch = /(^|\s)@([a-zA-Z0-9_]*)$/.exec(before)
   269→    if (mentionMatch) {
   270→      setMenu({ kind: 'mention', query: mentionMatch[2], start: caret - mentionMatch[2].length - 1 })
   271→      setMenuIndex(0)
   272→      return
   273→    }
   274→    const channelMatch = /(^|\s)#([a-z0-9-]*)$/i.exec(before)
   275→    if (channelMatch) {
   276→      setMenu({ kind: 'channel', query: channelMatch[2], start: caret - channelMatch[2].length - 1 })
   277→      setMenuIndex(0)
   278→      return
   279→    }
   280→    setMenu(null)
   281→  }
   282→
   283→  const handleChange = (value: string) => {
   284→    setText(value)
   285→    if (!editingMessage) setDraft(draftKey, value)
   286→    const el = textareaRef.current
   287→    detectMenu(value, el?.selectionStart ?? value.length)
   288→
   289→    if (!editingMessage && value.trim()) {
   290→      const now = Date.now()
   291→      if (now - lastTypingEmit.current > 2000) {
   292→        lastTypingEmit.current = now
   293→        if (channelId) emitTyping(channelId)
   294→      }
   295→    }
   296→  }
   297→
   298→  const insertAtCaret = (insert: string) => {
   299→    const el = textareaRef.current
   300→    if (!el) return
   301→    const start = el.selectionStart
   302→    const end = el.selectionEnd
   303→    const next = text.slice(0, start) + insert + text.slice(end)
   304→    handleChange(next)
   305→    requestAnimationFrame(() => {
   306→      const caret = start + insert.length
   307→      el.focus()
   308→      el.setSelectionRange(caret, caret)
   309→    })
   310→  }
   311→
   312→  const wrapSelection = (before: string, after = before, placeholderText = 'text') => {
   313→    const el = textareaRef.current
   314→    if (!el) return
   315→    const start = el.selectionStart
   316→    const end = el.selectionEnd
   317→    const selected = text.slice(start, end) || placeholderText
   318→    const next = text.slice(0, start) + before + selected + after + text.slice(end)
   319→    handleChange(next)
   320→    requestAnimationFrame(() => {
   321→      el.focus()
   322→      el.setSelectionRange(start + before.length, start + before.length + selected.length)
   323→    })
   324→  }
   325→
   326→  const selectMention = (selection: UserDTO | 'channel' | 'here') => {
   327→    if (!menu) return
   328→    const el = textareaRef.current
   329→    const label =
   330→      selection === 'channel' ? '@channel' : selection === 'here' ? '@here' : `@${selection.name}`
   331→    const value = text.slice(0, menu.start) + label + ' '
   332→    const next = value + text.slice((el?.selectionStart ?? menu.start + menu.query.length + 1))
   333→    setMenu(null)
   334→    handleChange(next)
   335→    requestAnimationFrame(() => {
   336→      const caret = value.length
   337→      el?.focus()
   338→      el?.setSelectionRange(caret, caret)
   339→    })
   340→  }
   341→
   342→  const selectChannel = (slug: string) => {
   343→    if (!menu) return
   344→    const el = textareaRef.current
   345→    const value = text.slice(0, menu.start) + `#${slug} `
   346→    const next = value + text.slice((el?.selectionStart ?? menu.start + menu.query.length + 1))
   347→    setMenu(null)
   348→    handleChange(next)
   349→    requestAnimationFrame(() => {
   350→      const caret = value.length
   351→      el?.focus()
   352→      el?.setSelectionRange(caret, caret)
   353→    })
   354→  }
   355→
   356→  const uploadFiles = async (list: FileList | File[]) => {
   357→    for (const file of Array.from(list)) {
   358→      if (file.size > 25 * 1024 * 1024) {
   359→        toast.error(`${file.name} is over the 25 MB limit`)
   360→        continue
   361→      }
   362→      setUploading((n) => n + 1)
   363→      try {
   364→        const uploaded = await uploadFile(file)
   365→        setFiles((prev) => [...prev, uploaded])
   366→      } catch (err) {
   367→        toast.error(err instanceof Error ? err.message : 'Upload failed')
   368→      } finally {
   369→        setUploading((n) => n - 1)
   370→      }
   371→    }
   372→  }
   373→
   374→  const submit = async () => {
   375→    const body = text.trim()
   376→    if (!body && files.length === 0) return
   377→    if (!body) return
   378→    if (!channelId) return
   379→    try {
   380→      if (editingMessage) {
   381→        await editMessage(editingMessage.id, body)
   382→        toast.success('Message updated')
   383→        return
   384→      }
   385→      await sendMessage(body, files.length > 0 ? files.map((f) => f.id) : undefined, parentId)
   386→      setText('')
   387→      setDraft(draftKey, '')
   388→      setFiles([])
   389→      requestAnimationFrame(() => {
   390→        const el = textareaRef.current
   391→        if (el) {
   392→          el.style.height = 'auto'
   393→          el.focus()
   394→        }
   395→      })
   396→    } catch (err) {
   397→      toast.error(err instanceof Error ? err.message : 'Could not send message')
   398→    }
   399→  }
   400→
   401→  const handleKeyDown = (event: React.KeyboardEvent<HTMLTextAreaElement>) => {
   402→    if (menu) {
   403→      if (event.key === 'ArrowDown') {
   404→        event.preventDefault()
   405→        setMenuIndex((i) => i + 1)
   406→        return
   407→      }
   408→      if (event.key === 'ArrowUp') {
   409→        event.preventDefault()
   410→        setMenuIndex((i) => Math.max(0, i - 1))
   411→        return
   412→      }
   413→      if (event.key === 'Enter' || event.key === 'Tab') {
   414→        event.preventDefault()
   415→        if (menu.kind === 'mention') {
   416→          const q = menu.query.toLowerCase()
   417→          const filtered = users.filter(
   418→            (user) => !q || user.name.toLowerCase().includes(q) || (user.handle ?? '').includes(q),
   419→          )
   420→          const offset = q.length >= 2 ? 2 : 0
   421→          if (menuIndex === 0 && q.length >= 2) selectMention('channel')
   422→          else if (menuIndex === 1 && q.length >= 2) selectMention('here')
   423→          else if (filtered[menuIndex - offset]) selectMention(filtered[menuIndex - offset])
   424→        } else {
   425→          const q = menu.query.toLowerCase()
   426→          const match = channels.find(
   427→            (c) => c.kind !== 'dm' && c.kind !== 'group_dm' && (c.slug.includes(q) || c.name.toLowerCase().includes(q)),
   428→          )
   429→          if (match) selectChannel(match.slug)
   430→        }
   431→        return
   432→      }
   433→      if (event.key === 'Escape') {
   434→        event.preventDefault()
   435→        setMenu(null)
   436→        return
   437→      }
   438→    }
   439→
   440→    if (event.key === 'Enter' && !event.shiftKey) {
   441→      event.preventDefault()
   442→      void submit()
   443→      return
   444→    }
   445→    if (event.key === 'Escape' && editingMessage) {
   446→      event.preventDefault()
   447→      setEditingMessageId(null)
   448→      setText(drafts[draftKey] ?? '')
   449→      return
   450→    }
   451→    if (event.key === 'ArrowUp' && !text && !editingMessage && !parentId) {
   452→      // edit my own last message in this channel
   453→      const mine = [...messages].reverse().find((m) => m.sender?.id === me?.id && !m.deletedAt)
   454→      if (mine) {
   455→        event.preventDefault()
   456→        setEditingMessageId(mine.id)
   457→      }
   458→    }
   459→  }
   460→
   461→  if (!channel || channel.isArchived) {
   462→    return (
   463→      <div className="shrink-0 px-4 pb-4 md:px-6">
   464→        <div className="rounded-xl border border-dashed border-border bg-muted/30 px-4 py-3 text-center text-sm text-muted-foreground">
   465→          This channel is archived — history is read-only.
   466→        </div>
   467→      </div>
   468→    )
   469→  }
   470→
   471→  const displayPlaceholder =
   472→    placeholder ??
   473→    (channel.kind === 'dm' || channel.kind === 'group_dm'
   474→      ? channel.members?.length === 1
   475→        ? `Message ${channel.members[0].name}${channel.members[0].kind === 'agent' ? ' — try asking anything' : ''}`
   476→        : 'Message group'
   477→      : `Message #${channel.name}`)
   478→
   479→  const mentionFiltered = useMemo(() => {
   480→    if (!menu || menu.kind !== 'mention') return []
   481→    const q = menu.query.toLowerCase()
   482→    return users
   483→      .filter((user) => !q || user.name.toLowerCase().includes(q) || (user.handle ?? '').toLowerCase().includes(q))
   484→      .slice(0, 7)
   485→  }, [menu, users])
   486→
   487→  const channelFiltered = useMemo(() => {
   488→    if (!menu || menu.kind !== 'channel') return []
   489→    const q = menu.query.toLowerCase()
   490→    return channels
   491→      .filter(
   492→        (c) =>
   493→          c.kind !== 'dm' && c.kind !== 'group_dm' && (c.slug.includes(q) || c.name.toLowerCase().includes(q)),
   494→      )
   495→      .slice(0, 7)
   496→  }, [menu, channels])
   497→
   498→  const toolbarButton =
   499→    'flex h-7 w-7 items-center justify-center rounded-md text-muted-foreground transition-colors duration-150 hover:bg-accent hover:text-foreground'
   500→
   501→  return (
   502→    <div className="relative shrink-0 px-4 pb-4 md:px-6">
   503→      {!parentId && channelId && <TypingIndicator channelId={channelId} />}
   504→
   505→      {/* edit bar */}
   506→      {editingMessage && (
   507→        <div className="mb-1.5 flex items-center gap-2 rounded-t-xl border border-b-0 border-amber-500/40 bg-amber-500/10 px-3 py-1.5 text-xs font-medium text-amber-700 dark:text-amber-300">
   508→          <span className="flex-1">
   509→            Editing message
   510→            <span className="ml-1.5 font-normal text-amber-600/70 dark:text-amber-400/70">Esc to cancel</span>
   511→          </span>
   512→          <button
   513→            type="button"
   514→            aria-label="Cancel edit"
   515→            onClick={() => {
   516→              setEditingMessageId(null)
   517→              setText(drafts[draftKey] ?? '')
   518→            }}
   519→            className="flex h-6 w-6 items-center justify-center rounded hover:bg-amber-500/20"
   520→          >
   521→            <X className="h-3.5 w-3.5" aria-hidden />
   522→          </button>
   523→          <button
   524→            type="button"
   525→            aria-label="Save edit"
   526→            onClick={() => void submit()}
   527→            className="flex h-6 w-6 items-center justify-center rounded hover:bg-amber-500/20"
   528→          >
   529→            <Check className="h-3.5 w-3.5" aria-hidden />
   530→          </button>
   531→        </div>
   532→      )}
   533→
   534→      <div
   535→        className={cn(
   536→          'relative rounded-xl border bg-card shadow-sm transition-colors duration-150 focus-within:border-emerald-500/50',
   537→          editingMessage ? 'rounded-t-none border-amber-500/40' : 'border-border',
   538→          dragging && 'border-emerald-500 ring-2 ring-emerald-500/30',
   539→        )}
   540→        onDragOver={(event) => {
   541→          event.preventDefault()
   542→          setDragging(true)
   543→        }}
   544→        onDragLeave={() => setDragging(false)}
   545→        onDrop={(event) => {
   546→          event.preventDefault()
   547→          setDragging(false)
   548→          if (event.dataTransfer.files.length > 0) void uploadFiles(event.dataTransfer.files)
   549→        }}
   550→      >
   551→        {/* drop overlay */}
   552→        {dragging && (
   553→          <div className="absolute inset-0 z-20 flex flex-col items-center justify-center gap-1 rounded-xl border-2 border-dashed border-emerald-500 bg-emerald-500/10 backdrop-blur-sm">

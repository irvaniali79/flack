    24→function loadDrafts(): Record<string, string> {
    25→  try {
    26→    const raw = localStorage.getItem(DRAFT_KEY)
    27→    const parsed = raw ? JSON.parse(raw) : {}
    28→    return typeof parsed === 'object' && parsed ? parsed : {}
    29→  } catch {
    30→    return {}
    31→  }
    32→}

    44→function loadSavedIds(): string[] {
    45→  try {
    46→    const raw = localStorage.getItem(SAVED_KEY)
    47→    const parsed = raw ? JSON.parse(raw) : []
    48→    return Array.isArray(parsed) ? parsed.filter((id) => typeof id === 'string') : []
    49→  } catch {
    50→    return []
    51→  }
    52→}

    62→/** Find a message DTO among the currently loaded channels/threads. */
    63→function findLoadedMessage(state: ChatState, id: string): MessageDTO | null {
    64→  for (const list of Object.values(state.messagesByChannel)) {
    65→    const hit = (list ?? []).find((m) => m.id === id)
    66→    if (hit) return hit
    67→  }
    68→  for (const replies of Object.values(state.threadReplies)) {
    69→    const hit = (replies ?? []).find((m) => m.id === id)
    70→    if (hit) return hit
    71→  }
    72→  return null
    73→}

   219→function replaceMessage(list: MessageDTO[], messageId: string, next: MessageDTO): MessageDTO[] {
   220→  const idx = list.findIndex((m) => m.id === messageId)
   221→  if (idx === -1) return list
   222→  const copy = [...list]
   223→  copy[idx] = next
   224→  return copy
   225→}
   226→
   227→function bumpReplyCount(list: MessageDTO[], rootId: string, delta: number): MessageDTO[] {
   228→  const idx = list.findIndex((m) => m.id === rootId)
   229→  if (idx === -1) return list
   230→  const copy = [...list]
   231→  copy[idx] = { ...copy[idx], replyCount: Math.max(0, copy[idx].replyCount + delta) }
   232→  return copy
   233→}

   235→function touchChannel(channels: ChannelDTO[], channelId: string, message: MessageDTO): ChannelDTO[] {
   236→  return channels.map((c) =>
   237→    c.id === channelId
   238→      ? {
   239→          ...c,
   240→          lastMessage: {
   241→            body: message.body,
   242→            createdAt: message.createdAt,
   243→            senderName: message.sender?.name ?? null,
   244→          },
   245→        }
   246→      : c,
   247→  )
   248→}

   399→  // ── channels ───────────────────────────────────────────────────────────────
   400→  openChannel: async (channelId) => {
   401→    const prev = get().activeChannelId
   402→    if (prev && prev !== channelId) socketLeaveChannel(prev)
   403→    set({ activeChannelId: channelId, activeThreadRootId: null, drawerOpen: false, editingMessageId: null })

   408→    await get().fetchMessages(channelId)
   409→
   410→    // Track the unread divider position before marking read
   411→    const messages = get().messagesByChannel[channelId] ?? []
   412→    let divider: string | null = null

   414→      const idx = Math.max(0, messages.length - unread)
   415→      divider = messages[idx]?.id ?? null
   416→    }
   417→    const last = messages[messages.length - 1]
   418→    if (last) get().markChannelRead(channelId, last.id)
   419→    set((s) => ({
   420→      unreadDividerByChannel: { ...s.unreadDividerByChannel, [channelId]: divider },
   421→    }))

   430→  fetchMessages: async (channelId) => {

   437→      set((s) => ({

   439→        hasMoreByChannel: { ...s.hasMoreByChannel, [channelId]: data.hasMore },
   440→        messagesLoading: { ...s.messagesLoading, [channelId]: false },
   441→      }))
   442→    } catch {
   443→      set((s) => ({ messagesLoading: { ...s.messagesLoading, [channelId]: false } }))
   444→    }

   447→  loadOlder: async (channelId) => {
   448→    const { messagesByChannel, hasMoreByChannel, loadingOlder } = get()
   449→    if (!hasMoreByChannel[channelId] || loadingOlder[channelId]) return

   451→    const oldest = messages[0]
   452→    if (!oldest) return
   453→    set((s) => ({ loadingOlder: { ...s.loadingOlder, [channelId]: true } }))
   454→    try {
   455→      const data = await api<{ messages: MessageDTO[]; hasMore: boolean }>(
   456→        `/api/channels/${channelId}/messages?limit=50&beforeId=${oldest.id}`,
   457→      )
   458→      set((s) => ({
   459→        messagesByChannel: {
   460→          ...s.messagesByChannel,
   461→          [channelId]: [...data.messages, ...(s.messagesByChannel[channelId] ?? [])],
   462→        },
   463→        hasMoreByChannel: { ...s.hasMoreByChannel, [channelId]: data.hasMore },
   464→        loadingOlder: { ...s.loadingOlder, [channelId]: false },
   465→      }))
   466→    } catch {
   467→      set((s) => ({ loadingOlder: { ...s.loadingOlder, [channelId]: false } }))
   468→    }
   469→  },

   471→  // ── messages ───────────────────────────────────────────────────────────────
   472→  sendMessage: async (body, fileIds, parentId) => {
   473→    const channelId = get().activeChannelId
   474→    if (!channelId) return
   475→    const data = await api<{ message: MessageDTO }>(`/api/channels/${channelId}/messages`, {
   476→      method: 'POST',
   477→      body: { body, fileIds, parentId },
   478→    })
   479→    const message = data.message
   480→    set((s) => {
   481→      if (message.parentId) {
   482→        const replies = s.threadReplies[message.parentId] ?? []
   483→        if (replies.some((r) => r.id === message.id)) return {}
   484→        return {
   485→          threadReplies: { ...s.threadReplies, [message.parentId]: [...replies, message] },
   486→          messagesByChannel: {
   487→            ...s.messagesByChannel,
   488→            [channelId]: bumpReplyCount(s.messagesByChannel[channelId] ?? [], message.parentId, 1),
   489→          },
   490→          channels: touchChannel(s.channels, channelId, message),
   491→          // Replying auto-follows the thread (server does this too — mirror locally)
   492→          threadFollowing: { ...s.threadFollowing, [message.parentId]: true },
   493→        }
   494→      }
   495→      const list = s.messagesByChannel[channelId] ?? []
   496→      if (list.some((m) => m.id === message.id)) return {}
   497→      return {
   498→        messagesByChannel: { ...s.messagesByChannel, [channelId]: [...list, message] },
   499→        channels: touchChannel(s.channels, channelId, message),
   500→      }
   501→    })
   502→    // Clear typing indicator for me
   503→    get().handleTyping(channelId, message.sender?.id ?? 'me', '', undefined, true)
   504→  },
   505→
   506→  editMessage: async (messageId, body) => {
   507→    const data = await api<{ message: MessageDTO }>(`/api/messages/${messageId}`, {
   508→      method: 'PATCH',
   509→      body: { body },
   510→    })
   511→    get().handleMessageUpdated(data.message)
   512→    set({ editingMessageId: null })
   513→  },
   514→
   515→  deleteMessage: async (messageId) => {
   516→    await api(`/api/messages/${messageId}`, { method: 'DELETE' })
   517→    const { messagesByChannel } = get()
   518→    const entry = Object.entries(messagesByChannel).find(([, list]) =>
   519→      list.some((m) => m.id === messageId),
   520→    )
   521→    const channelId = entry?.[0]
   522→    if (channelId) get().handleMessageDeleted(messageId, channelId)
   523→    // also drop from thread replies if present
   524→    set((s) => {
   525→      const nextThreadReplies: Record<string, MessageDTO[]> = {}
   526→      for (const [rootId, replies] of Object.entries(s.threadReplies)) {
   527→        nextThreadReplies[rootId] = replies.map((r) =>
   528→          r.id === messageId ? { ...r, deletedAt: new Date().toISOString(), isPinned: false } : r,
   529→        )
   530→      }
   531→      return { threadReplies: nextThreadReplies }
   532→    })
   533→  },
   534→
   535→  toggleReaction: async (messageId, emoji) => {
   536→    const data = await api<{ reactions: ReactionDTO[] }>(`/api/messages/${messageId}/reactions`, {
   537→      method: 'POST',
   538→      body: { emoji },
   539→    })
   540→    const { messagesByChannel, threadReplies } = get()
   541→    for (const [channelId] of Object.entries(messagesByChannel)) {
   542→      if (messagesByChannel[channelId]?.some((m) => m.id === messageId)) {
   543→        get().handleReactionUpdated(channelId, messageId, data.reactions)
   544→      }
   545→    }
   546→    for (const [rootId, replies] of Object.entries(threadReplies)) {
   547→      if (replies.some((m) => m.id === messageId)) {
   548→        set((s) => ({

   843→  handleNewMessage: (message) => {
   844→    const me = get().me
   845→    set((s) => {
   846→      const isMine = !!me && message.sender?.id === me.id
   847→      // Thread reply
   848→      if (message.parentId) {
   849→        const replies = s.threadReplies[message.parentId]
   850→        const channels = touchChannel(s.channels, message.channelId, message)
   851→        if (!replies) {
   852→          return {
   853→            channels,
   854→            messagesByChannel: {
   855→              ...s.messagesByChannel,
   856→              [message.channelId]: bumpReplyCount(
   857→                s.messagesByChannel[message.channelId] ?? [],
   858→                message.parentId,
   859→                1,
   860→              ),
   861→            },
   862→          }
   863→        }
   864→        if (replies.some((r) => r.id === message.id)) return {}
   865→        return {
   866→          channels,
   867→          threadReplies: { ...s.threadReplies, [message.parentId]: [...replies, message] },
   868→          messagesByChannel: {
   869→            ...s.messagesByChannel,
   870→            [message.channelId]: bumpReplyCount(
   871→              s.messagesByChannel[message.channelId] ?? [],
   872→              message.parentId,
   873→              1,
   874→            ),
   875→          },
   876→        }
   877→      }
   878→
   879→      // Top-level message
   880→      const list = s.messagesByChannel[message.channelId]
   881→      const channels = touchChannel(s.channels, message.channelId, message)
   882→      if (!list) {
   883→        // Channel not loaded — bump unread counters instead
   884→        return {
   885→          channels: isMine
   886→            ? channels
   887→            : channels.map((c) =>
   888→                c.id === message.channelId && c.isMember
   889→                  ? { ...c, unread: c.unread + 1 }
   890→                  : c,
   891→              ),
   892→        }
   893→      }
   894→      if (list.some((m) => m.id === message.id)) return {}
   895→      const isActive = s.activeChannelId === message.channelId
   896→      const shouldCountUnread = !isMine && !isActive
   897→      return {
   898→        channels: shouldCountUnread
   899→          ? channels.map((c) =>
   900→              c.id === message.channelId && c.isMember ? { ...c, unread: c.unread + 1 } : c,
   901→            )
   902→          : channels,
   903→        messagesByChannel: { ...s.messagesByChannel, [message.channelId]: [...list, message] },
   904→      }
   905→    })
   906→    // Clear the sender's typing indicator
   907→    if (message.sender) {
   908→      get().handleTyping(message.channelId, message.sender.id, '', undefined, true)
   909→    }
   910→    // If channel is active but window hidden, mark unread for badge (skip — keep simple)
   911→  },

   999→export function channelDisplayName(channel: ChannelDTO | null, meId: string | null): string {
  1000→  if (!channel) return ''
  1001→  if (channel.kind === 'dm') {
  1002→    const others = channel.members ?? []
  1003→    if (others.length === 1) return others[0].name
  1004→    return 'Direct message'
  1005→  }
  1006→  if (channel.kind === 'group_dm') {
  1007→    const others = (channel.members ?? []).map((m) => m.name.split(' ')[0])
  1008→    return others.join(', ') || channel.name
  1009→  }
  1010→  return channel.name
  1011→}

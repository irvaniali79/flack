    38→/** Live server status — pings the MCP endpoint with the session cookie. */
    39→function ServerStatus() {
    40→  const [status, setStatus] = useState<'checking' | 'online' | 'offline'>('checking')
    41→
    42→  useEffect(() => {
    43→    let cancelled = false
    44→    const ping = async () => {
    45→      try {
    46→        const res = await fetch('/api/mcp', {
    47→          method: 'POST',
    48→          headers: { 'Content-Type': 'application/json' },
    49→          body: JSON.stringify({ jsonrpc: '2.0', id: 1, method: 'ping' }),
    50→        })
    51→        const data = (await res.json()) as { result?: unknown; error?: unknown }
    52→        if (!cancelled) setStatus(data.result !== undefined ? 'online' : 'offline')
    53→      } catch {
    54→        if (!cancelled) setStatus('offline')
    55→      }
    56→    }
    57→    void ping()
    58→    const onFocus = () => void ping()
    59→    window.addEventListener('focus', onFocus)
    60→    return () => {
    61→      cancelled = true
    62→      window.removeEventListener('focus', onFocus)
    63→    }
    64→  }, [])
    65→
    66→  if (status === 'checking') {
    67→    return (
    68→      <Badge variant="outline" className="h-6 gap-1.5 px-2 text-[11px] text-muted-foreground">
    69→        <Loader2 className="h-3 w-3 animate-spin" aria-hidden /> checking…
    70→      </Badge>
    71→    )
    72→  }
    73→  if (status === 'online') {
    74→    return (
    75→      <Badge variant="outline" className="h-6 gap-1.5 border-emerald-500/40 bg-emerald-500/10 px-2 text-[11px] font-medium text-emerald-600 dark:text-emerald-400">
    76→        <span className="relative flex h-2 w-2">
    77→          <span className="absolute inline-flex h-full w-full animate-ping rounded-full bg-emerald-400 opacity-60" />
    78→          <span className="relative inline-flex h-2 w-2 rounded-full bg-emerald-500" />
    79→        </span>
    80→        MCP server online
    81→      </Badge>
    82→    )
    83→  }
    84→  return (
    85→    <Badge variant="outline" className="h-6 gap-1.5 border-destructive/40 px-2 text-[11px] text-destructive">
    86→      <CircleDot className="h-3 w-3" aria-hidden /> offline
    87→    </Badge>
    88→  )
    89→}

   415→function SamplingSection({ origin }: { origin: string }) {

   424→    setBusy(true)

     3→import { toast } from 'sonner'
     4→import {
     5→  ArrowDown,
     6→  ArrowUp,
     7→  Bot,
     8→  GripVertical,
     9→  MessageSquarePlus,
    10→  Plus,
    11→  Send,
    12→  SmilePlus,
    13→  X,
    14→  Zap,
    15→} from 'lucide-react'
    16→import {
    17→  Dialog,
    18→  DialogContent,
    19→  DialogDescription,
    20→  DialogFooter,
    21→  DialogHeader,
    22→  DialogTitle,
    23→} from '@/components/ui/dialog'
    24→import {
    25→  DropdownMenu,
    26→  DropdownMenuContent,
    27→  DropdownMenuItem,
    28→  DropdownMenuTrigger,
    29→} from '@/components/ui/dropdown-menu'
    30→import { Button } from '@/components/ui/button'
    31→import { Input } from '@/components/ui/input'
    32→import { Label } from '@/components/ui/label'
    33→import { Textarea } from '@/components/ui/textarea'
    34→import {
    35→  Select,
    36→  SelectContent,
    37→  SelectItem,
    38→  SelectTrigger,
    39→  SelectValue,
    40→} from '@/components/ui/select'
    41→import { api } from '@/lib/api'
    42→import { useChatStore } from '@/lib/store'
    43→import type { WorkflowDTO, WorkflowStep, WorkflowStepType, WorkflowTriggerType } from '@/lib/types'
    44→import { humanizeTrigger } from './trigger-label'
    45→
    46→const EMOJI_QUICK_PICKS = ['👍', '✅', '🎉', '❤️', '🚀', '👀', '😄', '🙏', '🙌', '⚡']
    47→const TEMPLATE_VARS = ['{{actor.name}}', '{{channel.name}}', '{{message.body}}', '{{message.id}}'] as const
    48→const ANY_CHANNEL = '__any__'
    49→const TRIGGER_CHANNEL = '__trigger__'
    50→
    51→const STEP_TYPES: { type: WorkflowStepType; label: string; icon: typeof Send }[] = [
    52→  { type: 'post_message', label: 'Post a message', icon: MessageSquarePlus },
    53→  { type: 'send_dm', label: 'Send a DM', icon: Send },
    54→  { type: 'add_reaction', label: 'Add a reaction', icon: SmilePlus },
    55→  { type: 'run_agent', label: 'Run an AI agent', icon: Bot },
    56→]
    57→
    58→interface BuilderForm {
    59→  name: string
    60→  description: string
    61→  triggerType: WorkflowTriggerType
    62→  triggerConfig: { channelId?: string; emoji?: string; keyword?: string }
    63→  steps: WorkflowStep[]
    64→}
    65→
    66→function emptyForm(): BuilderForm {
    67→  return {
    68→    name: '',
    69→    description: '',
    70→    triggerType: 'message_posted',
    71→    triggerConfig: {},
    72→    steps: [],
    73→  }
    74→}
    75→
    76→function formFromWorkflow(workflow: WorkflowDTO): BuilderForm {
    77→  return {
    78→    name: workflow.name,
    79→    description: workflow.description ?? '',
    80→    triggerType: workflow.triggerType,
    81→    triggerConfig: { ...workflow.triggerConfig },
    82→    steps: workflow.steps.map((step) => ({
    83→      ...step,
    84→      config: { ...step.config },
    85→      condition: step.condition ? { ...step.condition } : undefined,
    86→    })),
    87→  }
    88→}
    89→
    90→export function WorkflowBuilder({
    91→  open,
    92→  onOpenChange,
    93→  workflow,
    94→  onSaved,
    95→}: {
    96→  open: boolean
    97→  onOpenChange: (open: boolean) => void
    98→  workflow: WorkflowDTO | null
    99→  onSaved: () => void
   100→}) {
   101→  const channels = useChatStore((s) => s.channels)
   102→  const users = useChatStore((s) => s.users)
   103→  const agents = useChatStore((s) => s.agents)
   104→
   105→  // Guarded render-time reset: reinitialize the form whenever the dialog opens
   106→  // for a different workflow (or for "new").
   107→  const [seed, setSeed] = useState<{ open: boolean; id: string | null }>({
   108→    open: false,
   109→    id: null,
   110→  })
   111→  const [form, setForm] = useState<BuilderForm>(emptyForm)
   112→  const [busy, setBusy] = useState(false)
   113→
   114→  const currentSeedId = workflow?.id ?? null
   115→  if (open !== seed.open || currentSeedId !== seed.id) {
   116→    setSeed({ open, id: currentSeedId })
   117→    setForm(workflow ? formFromWorkflow(workflow) : emptyForm())
   118→  }
   119→
   120→  const namedChannels = channels.filter((c) => c.kind === 'public' || c.kind === 'private')
   121→  const humanMembers = users.filter((u) => u.isActive)
   122→  const preview = humanizeTrigger(form.triggerType, form.triggerConfig, channels)
   123→
   124→  const patch = (partial: Partial<BuilderForm>) => setForm((prev) => ({ ...prev, ...partial }))
   125→  const patchStep = (index: number, partial: Partial<WorkflowStep>) =>
   126→    setForm((prev) => ({
   127→      ...prev,
   128→      steps: prev.steps.map((step, i) => (i === index ? { ...step, ...partial } : step)),
   129→    }))
   130→  const patchStepConfig = (index: number, partial: Partial<WorkflowStep['config']>) =>
   131→    setForm((prev) => ({
   132→      ...prev,
   133→      steps: prev.steps.map((step, i) =>
   134→        i === index ? { ...step, config: { ...step.config, ...partial } } : step,
   135→      ),
   136→    }))
   137→
   138→  const addStep = (type: WorkflowStepType) =>
   139→    setForm((prev) => ({
   140→      ...prev,
   141→      steps: [
   142→        ...prev.steps,
   143→        {
   144→          id: crypto.randomUUID(),
   145→          type,
   146→          config: type === 'add_reaction' ? { targetTriggerMessage: true } : {},
   147→        },
   148→      ],
   149→    }))
   150→
   151→  const removeStep = (index: number) =>
   152→    setForm((prev) => ({ ...prev, steps: prev.steps.filter((_, i) => i !== index) }))
   153→
   154→  const moveStep = (index: number, dir: -1 | 1) =>
   155→    setForm((prev) => {
   156→      const target = index + dir
   157→      if (target < 0 || target >= prev.steps.length) return prev
   158→      const steps = [...prev.steps]
   159→      ;[steps[index], steps[target]] = [steps[target], steps[index]]
   160→      return { ...prev, steps }
   161→    })
   162→
   163→  const insertTemplate = (index: number, variable: string) =>
   164→    setForm((prev) => ({
   165→      ...prev,
   166→      steps: prev.steps.map((step, i) =>
   167→        i === index
   168→          ? {
   169→              ...step,
   170→              config: {
   171→                ...step.config,
   172→                ...(step.config.body !== undefined
   173→                  ? { body: `${step.config.body ?? ''}${step.config.body ? ' ' : ''}${variable}` }
   174→                  : { prompt: `${step.config.prompt ?? ''}${step.config.prompt ? ' ' : ''}${variable}` }),
   175→              },
   176→            }
   177→          : step,
   178→      ),
   179→    }))
   180→
   181→  const validate = (): string | null => {
   182→    if (!form.name.trim()) return 'Give the workflow a name'
   183→    for (const [index, step] of form.steps.entries()) {
   184→      const n = index + 1
   185→      if (step.type === 'post_message' && !step.config.body?.trim())
   186→        return `Step ${n}: message body is empty`
   187→      if (step.type === 'send_dm') {
   188→        if (!step.config.userId) return `Step ${n}: pick a DM recipient`
   189→        if (!step.config.body?.trim()) return `Step ${n}: DM body is empty`
   190→      }
   191→      if (step.type === 'add_reaction' && !step.config.emoji?.trim())
   192→        return `Step ${n}: pick an emoji to react with`
   193→      if (step.type === 'run_agent' && !step.config.agentId) return `Step ${n}: pick an agent to run`
   194→    }
   195→    return null
   196→  }
   197→
   198→  const save = async () => {
   199→    if (busy) return
   200→    const problem = validate()
   201→    if (problem) {
   202→      toast.error(problem)
   203→      return
   204→    }
   205→    setBusy(true)
   206→    try {
   207→      const payload = {
   208→        name: form.name.trim(),
   209→        description: form.description.trim() || null,
   210→        triggerType: form.triggerType,
   211→        triggerConfig: form.triggerConfig,
   212→        steps: form.steps.map((step) => ({
   213→          id: step.id,
   214→          type: step.type,
   215→          label: step.label?.trim() || undefined,
   216→          config: step.config,
   217→          condition: step.condition?.keyword?.trim()
   218→            ? { keyword: step.condition.keyword.trim() }
   219→            : undefined,
   220→        })),
   221→      }
   222→      if (workflow) {
   223→        await api(`/api/workflows/${workflow.id}`, { method: 'PATCH', body: payload })
   224→        toast.success(`Updated “${payload.name}”`)
   225→      } else {
   226→        await api('/api/workflows', { method: 'POST', body: payload })
   227→        toast.success(`Created “${payload.name}”`)
   228→      }
   229→      onSaved()
   230→      onOpenChange(false)
   231→    } catch (err) {
   232→      toast.error(err instanceof Error ? err.message : 'Could not save workflow')
   233→    } finally {
   234→      setBusy(false)
   235→    }
   236→  }
   237→
   238→  return (
   239→    <Dialog open={open} onOpenChange={onOpenChange}>
   240→      <DialogContent className="flex max-h-[88vh] flex-col gap-0 overflow-hidden rounded-2xl p-0 sm:max-w-2xl">
   241→        <div className="min-h-0 flex-1 overflow-y-auto px-5 py-4 md:px-6">
   242→          <DialogHeader className="text-left">
   243→            <DialogTitle className="flex items-center gap-2">
   244→              <Zap className="h-4 w-4 text-emerald-600" aria-hidden />
   245→              {workflow ? 'Edit workflow' : 'New workflow'}
   246→            </DialogTitle>
   247→            <DialogDescription>
   248→              Pick a trigger, then chain steps. Use template vars like{' '}
   249→              <code className="rounded bg-muted px-1 text-[11px]">{'{{actor.name}}'}</code> in
   250→              message bodies.
   251→            </DialogDescription>
   252→          </DialogHeader>
   253→
   254→          <div className="mt-5 space-y-5">
   255→            {/* basics */}
   256→            <div className="grid gap-4 sm:grid-cols-2">
   257→              <div className="space-y-1.5">
   258→                <Label htmlFor="workflow-name">Name</Label>
   259→                <Input
   260→                  id="workflow-name"
   261→                  value={form.name}
   262→                  onChange={(event) => patch({ name: event.target.value })}
   263→                  placeholder="e.g. Celebrate deploys 🎉"
   264→                  className="rounded-lg"
   265→                />
   266→              </div>
   267→              <div className="space-y-1.5">
   268→                <Label htmlFor="workflow-description">
   269→                  Description <span className="font-normal text-muted-foreground">(optional)</span>
   270→                </Label>
   271→                <Input
   272→                  id="workflow-description"
   273→                  value={form.description}
   274→                  onChange={(event) => patch({ description: event.target.value })}
   275→                  placeholder="What does this automation do?"
   276→                  className="rounded-lg"
   277→                />
   278→              </div>
   279→            </div>
   280→
   281→            {/* trigger */}
   282→            <section className="space-y-3 rounded-xl border border-border p-4">
   283→              <div className="flex items-center gap-2">
   284→                <span className="flex h-6 w-6 items-center justify-center rounded-md bg-emerald-600/15 text-emerald-600 dark:text-emerald-400">
   285→                  <Zap className="h-3.5 w-3.5" aria-hidden />
   286→                </span>
   287→                <h3 className="text-sm font-semibold">Trigger</h3>
   288→              </div>
   289→              <div className="grid gap-3 sm:grid-cols-2">
   290→                <div className="space-y-1.5">
   291→                  <Label>When…</Label>
   292→                  <Select
   293→                    value={form.triggerType}
   294→                    onValueChange={(value) =>
   295→                      patch({ triggerType: value as WorkflowTriggerType })
   296→                    }
   297→                  >
   298→                    <SelectTrigger className="w-full rounded-lg" aria-label="Trigger type">
   299→                      <SelectValue />
   300→                    </SelectTrigger>
   301→                    <SelectContent className="rounded-xl">
   302→                      <SelectItem value="message_posted">A message is posted</SelectItem>
   303→                      <SelectItem value="reaction">A reaction is added</SelectItem>
   304→                      <SelectItem value="button">Someone clicks Run now</SelectItem>
   305→                      <SelectItem value="webhook">A webhook is received</SelectItem>
   306→                    </SelectContent>
   307→                  </Select>
   308→                </div>
   309→
   310→                {form.triggerType === 'message_posted' && (
   311→                  <>
   312→                    <div className="space-y-1.5">
   313→                      <Label>Channel</Label>
   314→                      <Select
   315→                        value={form.triggerConfig.channelId ?? ANY_CHANNEL}
   316→                        onValueChange={(value) =>
   317→                          patch({
   318→                            triggerConfig: {
   319→                              ...form.triggerConfig,
   320→                              channelId: value === ANY_CHANNEL ? undefined : value,
   321→                            },
   322→                          })
   323→                        }
   324→                      >
   325→                        <SelectTrigger className="w-full rounded-lg" aria-label="Trigger channel">
   326→                          <SelectValue />
   327→                        </SelectTrigger>
   328→                        <SelectContent className="rounded-xl">
   329→                          <SelectItem value={ANY_CHANNEL}>Any channel</SelectItem>
   330→                          {namedChannels.map((channel) => (
   331→                            <SelectItem key={channel.id} value={channel.id}>
   332→                              #{channel.name}
   333→                            </SelectItem>
   334→                          ))}
   335→                        </SelectContent>
   336→                      </Select>
   337→                    </div>
   338→                    <div className="space-y-1.5 sm:col-span-2">
   339→                      <Label htmlFor="trigger-keyword">
   340→                        Keyword <span className="font-normal text-muted-foreground">(optional)</span>
   341→                      </Label>
   342→                      <Input
   343→                        id="trigger-keyword"
   344→                        value={form.triggerConfig.keyword ?? ''}
   345→                        onChange={(event) =>
   346→                          patch({
   347→                            triggerConfig: { ...form.triggerConfig, keyword: event.target.value },
   348→                          })
   349→                        }
   350→                        placeholder="Only when the message contains… e.g. deployed"
   351→                        className="rounded-lg"
   352→                      />
   353→                    </div>
   354→                  </>
   355→                )}
   356→
   357→                {form.triggerType === 'reaction' && (
   358→                  <div className="space-y-1.5 sm:col-span-2">
   359→                    <Label htmlFor="trigger-emoji">
   360→                      Emoji <span className="font-normal text-muted-foreground">(blank = any)</span>
   361→                    </Label>
   362→                    <Input
   363→                      id="trigger-emoji"
   364→                      value={form.triggerConfig.emoji ?? ''}
   365→                      onChange={(event) =>
   366→                        patch({
   367→                          triggerConfig: { ...form.triggerConfig, emoji: event.target.value },
   368→                        })
   369→                      }
   370→                      placeholder="✅"
   371→                      className="w-24 rounded-lg"
   372→                    />
   373→                    <div className="flex flex-wrap gap-1 pt-1">
   374→                      {EMOJI_QUICK_PICKS.map((emoji) => (
   375→                        <button
   376→                          key={emoji}
   377→                          type="button"
   378→                          aria-label={`Use ${emoji}`}
   379→                          onClick={() =>
   380→                            patch({ triggerConfig: { ...form.triggerConfig, emoji } })
   381→                          }
   382→                          className="rounded-lg border border-border px-1.5 py-0.5 text-base leading-relaxed transition-colors duration-150 hover:border-emerald-500/50 hover:bg-emerald-500/5"
   383→                        >
   384→                          {emoji}
   385→                        </button>
   386→                      ))}
   387→                    </div>
   388→                  </div>
   389→                )}
   390→              </div>
   391→              <p className="rounded-lg bg-emerald-500/5 px-3 py-2 text-xs italic text-emerald-700 dark:bg-emerald-500/10 dark:text-emerald-300">
   392→                {preview}…
   393→              </p>
   394→            </section>
   395→
   396→            {/* steps */}
   397→            <section className="space-y-3">
   398→              <div className="flex items-center justify-between">
   399→                <h3 className="text-sm font-semibold">
   400→                  Steps{' '}
   401→                  <span className="font-normal text-muted-foreground">
   402→                    ({form.steps.length})
   403→                  </span>
   404→                </h3>
   405→                <DropdownMenu>
   406→                  <DropdownMenuTrigger asChild>
   407→                    <Button
   408→                      variant="outline"
   409→                      size="sm"
   410→                      className="rounded-lg border-emerald-600/40 text-emerald-700 hover:bg-emerald-600/10 hover:text-emerald-700 dark:text-emerald-400 dark:hover:text-emerald-400"
   411→                    >
   412→                      <Plus className="h-4 w-4" aria-hidden /> Add step
   413→                    </Button>
   414→                  </DropdownMenuTrigger>
   415→                  <DropdownMenuContent align="end" className="w-48 rounded-xl">
   416→                    {STEP_TYPES.map(({ type, label, icon: Icon }) => (
   417→                      <DropdownMenuItem key={type} onClick={() => addStep(type)}>
   418→                        <Icon className="h-4 w-4 text-muted-foreground" aria-hidden />
   419→                        {label}
   420→                      </DropdownMenuItem>
   421→                    ))}
   422→                  </DropdownMenuContent>
   423→                </DropdownMenu>
   424→              </div>
   425→
   426→              {form.steps.length === 0 ? (
   427→                <div className="rounded-xl border border-dashed border-border px-4 py-6 text-center text-xs text-muted-foreground">
   428→                  No steps yet — add what should happen when the trigger fires.
   429→                </div>
   430→              ) : (
   431→                <ol className="space-y-3">
   432→                  {form.steps.map((step, index) => (
   433→                    <li
   434→                      key={step.id}
   435→                      className="rounded-xl border border-border p-3.5"
   436→                    >
   437→                      <div className="flex items-center gap-2">
   438→                        <GripVertical
   439→                          className="h-4 w-4 shrink-0 text-muted-foreground/60"
   440→                          aria-hidden
   441→                        />
   442→                        <span className="flex h-6 w-6 items-center justify-center rounded-md bg-muted text-xs font-bold text-muted-foreground">
   443→                          {index + 1}
   444→                        </span>
   445→                        <Input
   446→                          value={step.label ?? ''}
   447→                          onChange={(event) => patchStep(index, { label: event.target.value })}
   448→                          placeholder="Step label (optional)"
   449→                          className="h-8 flex-1 rounded-lg border-0 bg-transparent px-1.5 text-sm shadow-none focus-visible:ring-1 focus-visible:ring-emerald-500"
   450→                          aria-label={`Step ${index + 1} label`}
   451→                        />
   452→                        <div className="flex items-center gap-0.5">
   453→                          <Button
   454→                            variant="ghost"
   455→                            size="icon"
   456→                            aria-label="Move step up"
   457→                            disabled={index === 0}
   458→                            onClick={() => moveStep(index, -1)}
   459→                            className="h-7 w-7 rounded-lg"
   460→                          >
   461→                            <ArrowUp className="h-3.5 w-3.5" aria-hidden />
   462→                          </Button>
   463→                          <Button
   464→                            variant="ghost"
   465→                            size="icon"
   466→                            aria-label="Move step down"
   467→                            disabled={index === form.steps.length - 1}
   468→                            onClick={() => moveStep(index, 1)}
   469→                            className="h-7 w-7 rounded-lg"
   470→                          >
   471→                            <ArrowDown className="h-3.5 w-3.5" aria-hidden />
   472→                          </Button>
   473→                          <Button
   474→                            variant="ghost"
   475→                            size="icon"
   476→                            aria-label="Remove step"
   477→                            onClick={() => removeStep(index)}
   478→                            className="h-7 w-7 rounded-lg text-muted-foreground hover:bg-rose-600/10 hover:text-rose-600 dark:hover:text-rose-400"
   479→                          >
   480→                            <X className="h-3.5 w-3.5" aria-hidden />
   481→                          </Button>
   482→                        </div>
   483→                      </div>
   484→
   485→                      <div className="mt-3 grid gap-3 sm:grid-cols-2">
   486→                        <div className="space-y-1.5">
   487→                          <Label>Type</Label>
   488→                          <Select
   489→                            value={step.type}
   490→                            onValueChange={(value) =>
   491→                              patchStep(index, {
   492→                                type: value as WorkflowStepType,
   493→                                config:
   494→                                  value === 'add_reaction' ? { targetTriggerMessage: true } : {},
   495→                              })
   496→                            }
   497→                          >
   498→                            <SelectTrigger className="w-full rounded-lg" aria-label="Step type">
   499→                              <SelectValue />
   500→                            </SelectTrigger>
   501→                            <SelectContent className="rounded-xl">
   502→                              {STEP_TYPES.map(({ type, label }) => (
   503→                                <SelectItem key={type} value={type}>
   504→                                  {label}
   505→                                </SelectItem>
   506→                              ))}
   507→                            </SelectContent>
   508→                          </Select>
   509→                        </div>
   510→
   511→                        {step.type === 'post_message' && (
   512→                          <div className="space-y-1.5">
   513→                            <Label>Channel</Label>
   514→                            <Select
   515→                              value={step.config.channelId ?? TRIGGER_CHANNEL}
   516→                              onValueChange={(value) =>
   517→                                patchStepConfig(index, {
   518→                                  channelId: value === TRIGGER_CHANNEL ? undefined : value,
   519→                                })
   520→                              }
   521→                            >
   522→                              <SelectTrigger className="w-full rounded-lg" aria-label="Post to channel">
   523→                                <SelectValue />
   524→                              </SelectTrigger>
   525→                              <SelectContent className="rounded-xl">
   526→                                <SelectItem value={TRIGGER_CHANNEL}>Where it was triggered</SelectItem>
   527→                                {namedChannels.map((channel) => (
   528→                                  <SelectItem key={channel.id} value={channel.id}>
   529→                                    #{channel.name}
   530→                                  </SelectItem>
   531→                                ))}
   532→                              </SelectContent>
   533→                            </Select>
   534→                          </div>
   535→                        )}
   536→
   537→                        {step.type === 'send_dm' && (
   538→                          <div className="space-y-1.5">
   539→                            <Label>Send to</Label>
   540→                            <Select
   541→                              value={step.config.userId ?? ''}
   542→                              onValueChange={(value) => patchStepConfig(index, { userId: value })}
   543→                            >
   544→                              <SelectTrigger className="w-full rounded-lg" aria-label="DM recipient">
   545→                                <SelectValue placeholder="Pick a member…" />
   546→                              </SelectTrigger>
   547→                              <SelectContent className="rounded-xl">
   548→                                {humanMembers.map((user) => (
   549→                                  <SelectItem key={user.id} value={user.id}>
   550→                                    {user.name}
   551→                                  </SelectItem>
   552→                                ))}
   553→                              </SelectContent>
   554→                            </Select>
   555→                          </div>
   556→                        )}
   557→
   558→                        {step.type === 'run_agent' && (
   559→                          <>
   560→                            <div className="space-y-1.5">
   561→                              <Label>Agent</Label>
   562→                              <Select

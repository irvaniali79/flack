    24→function loadDrafts(): Record<string, string> {
    25→  try {
    26→    const raw = localStorage.getItem(DRAFT_KEY)
    27→    const parsed = raw ? JSON.parse(raw) : {}
    28→    return typeof parsed === 'object' && parsed ? parsed : {}
    29→  } catch {
    30→    return {}
    31→  }
    32→}

    44→function loadSavedIds(): string[] {

    48→    return Array.isArray(parsed) ? parsed.filter((id) => typeof id === 'string') : []
    49→  } catch {
    50→    return []

    62→/** Find a message DTO among the currently loaded channels/threads. */
    63→function findLoadedMessage(state: ChatState, id: string): MessageDTO | null {
    64→  for (const list of Object.values(state.messagesByChannel)) {
    65→    const hit = (list ?? []).find((m) => m.id === id)
    66→    if (hit) return hit
    67→  }
    68→  for (const replies of Object.values(state.threadReplies)) {
    69→    const hit = (replies ?? []).find((m) => m.id === id)
    70→    if (hit) return hit
    71→  }
    72→  return null
    73→}

   206→function replaceMessage(list: MessageDTO[], messageId: string, next: MessageDTO): MessageDTO[] {
   207→  const idx = list.findIndex((m) => m.id === messageId)
   208→  if (idx === -1) return list
   209→  const copy = [...list]
   210→  copy[idx] = next
   211→  return copy
   212→}
   213→
   214→function bumpReplyCount(list: MessageDTO[], rootId: string, delta: number): MessageDTO[] {
   215→  const idx = list.findIndex((m) => m.id === rootId)
   216→  if (idx === -1) return list
   217→  const copy = [...list]
   218→  copy[idx] = { ...copy[idx], replyCount: Math.max(0, copy[idx].replyCount + delta) }
   219→  return copy
   220→}

   222→function touchChannel(channels: ChannelDTO[], channelId: string, message: MessageDTO): ChannelDTO[] {
   223→  return channels.map((c) =>
   224→    c.id === channelId
   225→      ? {
   226→          ...c,
   227→          lastMessage: {
   228→            body: message.body,
   229→            createdAt: message.createdAt,
   230→            senderName: message.sender?.name ?? null,
   231→          },
   232→        }
   233→      : c,
   234→  )
   235→}

   506→          threadReplies: {
   507→            ...s.threadReplies,
   508→            [rootId]: replies.map((r) => (r.id === messageId ? { ...r, reactions: data.reactions } : r)),
   509→          },
   510→        }))

   671→        for (const entry of results) if (entry) extras[entry[0]] = entry[1]

   714→  handleNewMessage: (message) => {
   715→    const me = get().me
   716→    set((s) => {
   717→      const isMine = !!me && message.sender?.id === me.id
   718→      // Thread reply
   719→      if (message.parentId) {
   720→        const replies = s.threadReplies[message.parentId]
   721→        const channels = touchChannel(s.channels, message.channelId, message)
   722→        if (!replies) {
   723→          return {
   724→            channels,
   725→            messagesByChannel: {
   726→              ...s.messagesByChannel,
   727→              [message.channelId]: bumpReplyCount(
   728→                s.messagesByChannel[message.channelId] ?? [],
   729→                message.parentId,
   730→                1,
   731→              ),
   732→            },
   733→          }
   734→        }
   735→        if (replies.some((r) => r.id === message.id)) return {}
   736→        return {
   737→          channels,
   738→          threadReplies: { ...s.threadReplies, [message.parentId]: [...replies, message] },
   739→          messagesByChannel: {
   740→            ...s.messagesByChannel,
   741→            [message.channelId]: bumpReplyCount(
   742→              s.messagesByChannel[message.channelId] ?? [],
   743→              message.parentId,
   744→              1,
   745→            ),
   746→          },
   747→        }
   748→      }
   749→
   750→      // Top-level message
   751→      const list = s.messagesByChannel[message.channelId]
   752→      const channels = touchChannel(s.channels, message.channelId, message)
   753→      if (!list) {
   754→        // Channel not loaded — bump unread counters instead
   755→        return {
   756→          channels: isMine
   757→            ? channels
   758→            : channels.map((c) =>
   759→                c.id === message.channelId && c.isMember
   760→                  ? { ...c, unread: c.unread + 1 }
   761→                  : c,
   762→              ),
   763→        }
   764→      }
   765→      if (list.some((m) => m.id === message.id)) return {}
   766→      const isActive = s.activeChannelId === message.channelId
   767→      const shouldCountUnread = !isMine && !isActive
   768→      return {
   769→        channels: shouldCountUnread
   770→          ? channels.map((c) =>
   771→              c.id === message.channelId && c.isMember ? { ...c, unread: c.unread + 1 } : c,
   772→            )
   773→          : channels,
   774→        messagesByChannel: { ...s.messagesByChannel, [message.channelId]: [...list, message] },
   775→      }
   776→    })
   777→    // Clear the sender's typing indicator
   778→    if (message.sender) {
   779→      get().handleTyping(message.channelId, message.sender.id, '', undefined, true)
   780→    }
   781→    // If channel is active but window hidden, mark unread for badge (skip — keep simple)
   782→  },

   867→export function channelDisplayName(channel: ChannelDTO | null, meId: string | null): string {
   868→  if (!channel) return ''
   869→  if (channel.kind === 'dm') {
   870→    const others = channel.members ?? []
   871→    if (others.length === 1) return others[0].name
   872→    return 'Direct message'
   873→  }
   874→  if (channel.kind === 'group_dm') {
   875→    const others = (channel.members ?? []).map((m) => m.name.split(' ')[0])
   876→    return others.join(', ') || channel.name
   877→  }
   878→  return channel.name
   879→}

    30→// ─── typing indicator ────────────────────────────────────────────────────────
    31→
    32→export function TypingIndicator({ channelId }: { channelId: string }) {
    33→  const typing = useChatStore((s) => s.typing[channelId])
    34→  const me = useChatStore((s) => s.me)
    35→  const [, setTick] = useState(0)
    36→
    37→  useEffect(() => {
    38→    const interval = setInterval(() => setTick((v) => v + 1), 1000)
    39→    return () => clearInterval(interval)
    40→  }, [])
    41→
    42→  const entries = Object.entries(typing ?? {})
    43→    .filter(([userId, entry]) => userId !== me?.id && Date.now() - entry.at < 4000)
    44→    .map(([, entry]) => entry)
    45→
    46→  if (entries.length === 0) return null
    47→  const agent = entries.find((entry) => entry.kind === 'agent')
    48→
    49→  return (
    50→    <div className="flex items-center gap-2 px-4 pb-1 text-xs text-muted-foreground md:px-5" aria-live="polite">
    51→      {agent ? (
    52→        <Sparkles className="h-3.5 w-3.5 text-amber-500" aria-hidden />
    53→      ) : (
    54→        <span className="flex gap-0.5" aria-hidden>
    55→          <span className="typing-dot h-1 w-1 rounded-full bg-emerald-500" />
    56→          <span className="typing-dot h-1 w-1 rounded-full bg-emerald-500" />
    57→          <span className="typing-dot h-1 w-1 rounded-full bg-emerald-500" />
    58→        </span>
    59→      )}
    60→      {entries.length === 1
    61→        ? `${entries[0].name} is ${agent ? 'thinking…' : 'typing…'}`
    62→        : agent
    63→          ? `${agent.name} is thinking…`
    64→          : `${entries.length} people are typing…`}
    65→    </div>
    66→  )
    67→}

    77→function MentionMenu({
    78→  users,
    79→  query,
    80→  onSelect,
    81→  onClose,
    82→  activeIndex,
    83→}: {
    84→  users: UserDTO[]
    85→  query: string
    86→  onSelect: (user: UserDTO | 'channel' | 'here') => void
    87→  onClose: () => void
    88→  activeIndex: number
    89→}) {
    90→  const q = query.toLowerCase()
    91→  const filtered = users
    92→    .filter((user) => !q || user.name.toLowerCase().includes(q) || (user.handle ?? '').toLowerCase().includes(q))
    93→    .slice(0, 7)
    94→
    95→  return (
    96→    <div className="absolute bottom-full left-0 z-30 mb-2 w-64 overflow-hidden rounded-xl border border-border bg-popover shadow-xl">
    97→      <p className="border-b border-border px-3 py-1.5 text-[10px] font-bold uppercase tracking-wider text-muted-foreground">
    98→        Mention
    99→      </p>
   100→      <div className="max-h-60 overflow-y-auto p-1">
   101→        {q.length >= 2 && (
   102→          <>
   103→            <MenuItem
   104→              label="@channel"
   105→              description="Notify everyone in this channel"
   106→              special
   107→              active={activeIndex === 0}
   108→              onSelect={() => onSelect('channel')}
   109→            />
   110→            <MenuItem
   111→              label="@here"
   112→              description="Notify people active now"
   113→              special
   114→              active={activeIndex === 1}
   115→              onSelect={() => onSelect('here')}
   116→            />
   117→          </>
   118→        )}
   119→        {filtered.map((user, index) => (
   120→          <button
   121→            key={user.id}
   122→            type="button"
   123→            className={cn(
   124→              'flex w-full items-center gap-2.5 rounded-lg px-2 py-1.5 text-left transition-colors duration-150',
   125→              activeIndex === index + (q.length >= 2 ? 2 : 0) ? 'bg-accent' : 'hover:bg-accent',
   126→            )}
   127→            onMouseDown={(event) => event.preventDefault()}
   128→            onClick={() => onSelect(user)}
   129→          >
   130→            <UserAvatar user={user} size="xs" />
   131→            <span className="min-w-0 flex-1">
   132→              <span className="block truncate text-sm">{user.name}</span>
   133→              {user.handle && (
   134→                <span className="block truncate text-[11px] text-muted-foreground">@{user.handle}</span>
   135→              )}
   136→            </span>
   137→            {user.kind === 'agent' && (
   138→              <span className="flex items-center gap-0.5 rounded bg-amber-500/15 px-1 py-px text-[9px] font-bold uppercase text-amber-600 dark:text-amber-400">
   139→                <Sparkles className="h-2.5 w-2.5" aria-hidden /> AI
   140→              </span>
   141→            )}
   142→          </button>
   143→        ))}
   144→        {filtered.length === 0 && q.length >= 2 && null}
   145→        {filtered.length === 0 && (
   146→          <p className="px-3 py-3 text-center text-xs text-muted-foreground">No matches</p>
   147→        )}
   148→      </div>
   149→      <button
   150→        type="button"
   151→        className="w-full border-t border-border px-3 py-1.5 text-left text-[11px] text-muted-foreground hover:bg-accent"
   152→        onMouseDown={(event) => event.preventDefault()}
   153→        onClick={onClose}
   154→      >
   155→        Esc to dismiss
   156→      </button>
   157→    </div>
   158→  )
   159→}

   201→// ─── composer ────────────────────────────────────────────────────────────────
   202→
   203→export function Composer({ parentId, placeholder }: { parentId?: string; placeholder?: string }) {
   204→  const channel = useChatStore(getActiveChannel)
   205→  const channelId = channel?.id ?? ''
   206→  const me = useChatStore((s) => s.me)
   207→  const users = useChatStore((s) => s.users)
   208→  const channels = useChatStore((s) => s.channels)
   209→  const sendMessage = useChatStore((s) => s.sendMessage)
   210→  const editMessage = useChatStore((s) => s.editMessage)
   211→  const emitTyping = useChatStore((s) => s.emitTyping)
   212→  const drafts = useChatStore((s) => s.drafts)
   213→  const setDraft = useChatStore((s) => s.setDraft)
   214→  const editingMessageId = useChatStore((s) => s.editingMessageId)
   215→  const setEditingMessageId = useChatStore((s) => s.setEditingMessageId)
   216→  const messages = useChatStore((s) => (channelId ? s.messagesByChannel[channelId] : undefined)) ?? []
   217→  const threadReplies = useChatStore((s) => (s.activeThreadRootId ? s.threadReplies[s.activeThreadRootId] : undefined))
   218→
   219→  const draftKey = parentId ? `${channelId}-thread-${parentId}` : channelId
   220→  const [text, setText] = useState('')
   221→  const [files, setFiles] = useState<UploadedFile[]>([])
   222→  const [uploading, setUploading] = useState(0)
   223→  const [menu, setMenu] = useState<MenuState | null>(null)
   224→  const [menuIndex, setMenuIndex] = useState(0)
   225→  const [dragging, setDragging] = useState(false)
   226→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   227→  const fileInputRef = useRef<HTMLInputElement>(null)
   228→  const lastTypingEmit = useRef(0)
   229→
   230→  // editing state: load message into the box
   231→  const editingMessage = useMemo(() => {
   232→    if (!editingMessageId) return null
   233→    const inChannel = messages.find((m) => m.id === editingMessageId)
   234→    const inThread = threadReplies?.find((m) => m.id === editingMessageId)
   235→    return inChannel ?? inThread ?? null
   236→  }, [editingMessageId, messages, threadReplies])
   237→
   238→  useEffect(() => {
   239→    if (editingMessage) {
   240→      setText(editingMessage.body)
   241→      textareaRef.current?.focus()
   242→    }
   243→  }, [editingMessage])
   244→
   245→  // restore draft when channel changes (skip when editing)
   246→  useEffect(() => {
   247→    if (!editingMessage) {
   248→      setText(drafts[draftKey] ?? '')
   249→    }
   250→    setFiles([])
   251→    setMenu(null)
   252→  }, [draftKey])
   253→
   254→  const isAgentDm = channel?.kind === 'dm' && channel.members?.some((m) => m.kind === 'agent')
   255→
   256→  const mentionFiltered = useMemo(() => {
   257→    if (!menu || menu.kind !== 'mention') return []
   258→    const q = menu.query.toLowerCase()
   259→    return users
   260→      .filter((user) => !q || user.name.toLowerCase().includes(q) || (user.handle ?? '').toLowerCase().includes(q))
   261→      .slice(0, 7)
   262→  }, [menu, users])
   263→
   264→  const channelFiltered = useMemo(() => {
   265→    if (!menu || menu.kind !== 'channel') return []
   266→    const q = menu.query.toLowerCase()
   267→    return channels
   268→      .filter(
   269→        (c) =>
   270→          c.kind !== 'dm' && c.kind !== 'group_dm' && (c.slug.includes(q) || c.name.toLowerCase().includes(q)),
   271→      )
   272→      .slice(0, 7)
   273→  }, [menu, channels])
   274→
   275→  const autosize = () => {
   276→    const el = textareaRef.current
   277→    if (!el) return
   278→    el.style.height = 'auto'
   279→    el.style.height = `${Math.min(el.scrollHeight, 160)}px`
   280→  }
   281→
   282→  useEffect(autosize, [text])
   283→
   284→  const detectMenu = (value: string, caret: number) => {
   285→    const before = value.slice(0, caret)
   286→    const mentionMatch = /(^|\s)@([a-zA-Z0-9_]*)$/.exec(before)
   287→    if (mentionMatch) {
   288→      setMenu({ kind: 'mention', query: mentionMatch[2], start: caret - mentionMatch[2].length - 1 })
   289→      setMenuIndex(0)
   290→      return
   291→    }
   292→    const channelMatch = /(^|\s)#([a-z0-9-]*)$/i.exec(before)
   293→    if (channelMatch) {
   294→      setMenu({ kind: 'channel', query: channelMatch[2], start: caret - channelMatch[2].length - 1 })
   295→      setMenuIndex(0)
   296→      return
   297→    }
   298→    setMenu(null)
   299→  }
   300→
   301→  const handleChange = (value: string) => {
   302→    setText(value)
   303→    if (!editingMessage) setDraft(draftKey, value)
   304→    const el = textareaRef.current
   305→    detectMenu(value, el?.selectionStart ?? value.length)
   306→
   307→    if (!editingMessage && value.trim()) {
   308→      const now = Date.now()
   309→      if (now - lastTypingEmit.current > 2000) {
   310→        lastTypingEmit.current = now
   311→        if (channelId) emitTyping(channelId)
   312→      }
   313→    }
   314→  }
   315→
   316→  const insertAtCaret = (insert: string) => {
   317→    const el = textareaRef.current
   318→    if (!el) return
   319→    const start = el.selectionStart
   320→    const end = el.selectionEnd
   321→    const next = text.slice(0, start) + insert + text.slice(end)
   322→    handleChange(next)
   323→    requestAnimationFrame(() => {
   324→      const caret = start + insert.length
   325→      el.focus()
   326→      el.setSelectionRange(caret, caret)
   327→    })
   328→  }
   329→
   330→  const wrapSelection = (before: string, after = before, placeholderText = 'text') => {
   331→    const el = textareaRef.current
   332→    if (!el) return
   333→    const start = el.selectionStart
   334→    const end = el.selectionEnd
   335→    const selected = text.slice(start, end) || placeholderText
   336→    const next = text.slice(0, start) + before + selected + after + text.slice(end)
   337→    handleChange(next)
   338→    requestAnimationFrame(() => {
   339→      el.focus()
   340→      el.setSelectionRange(start + before.length, start + before.length + selected.length)
   341→    })
   342→  }
   343→
   344→  const selectMention = (selection: UserDTO | 'channel' | 'here') => {
   345→    if (!menu) return
   346→    const el = textareaRef.current
   347→    const label =
   348→      selection === 'channel' ? '@channel' : selection === 'here' ? '@here' : `@${selection.name}`
   349→    const value = text.slice(0, menu.start) + label + ' '
   350→    const next = value + text.slice((el?.selectionStart ?? menu.start + menu.query.length + 1))
   351→    setMenu(null)
   352→    handleChange(next)
   353→    requestAnimationFrame(() => {
   354→      const caret = value.length
   355→      el?.focus()
   356→      el?.setSelectionRange(caret, caret)
   357→    })
   358→  }
   359→
   360→  const selectChannel = (slug: string) => {
   361→    if (!menu) return
   362→    const el = textareaRef.current
   363→    const value = text.slice(0, menu.start) + `#${slug} `
   364→    const next = value + text.slice((el?.selectionStart ?? menu.start + menu.query.length + 1))
   365→    setMenu(null)
   366→    handleChange(next)
   367→    requestAnimationFrame(() => {
   368→      const caret = value.length
   369→      el?.focus()
   370→      el?.setSelectionRange(caret, caret)
   371→    })
   372→  }
   373→
   374→  const uploadFiles = async (list: FileList | File[]) => {
   375→    for (const file of Array.from(list)) {
   376→      if (file.size > 25 * 1024 * 1024) {
   377→        toast.error(`${file.name} is over the 25 MB limit`)
   378→        continue
   379→      }
   380→      setUploading((n) => n + 1)
   381→      try {
   382→        const uploaded = await uploadFile(file)
   383→        setFiles((prev) => [...prev, uploaded])
   384→      } catch (err) {
   385→        toast.error(err instanceof Error ? err.message : 'Upload failed')
   386→      } finally {
   387→        setUploading((n) => n - 1)
   388→      }
   389→    }
   390→  }
   391→
   392→  const submit = async () => {
   393→    const body = text.trim()
   394→    if (!body && files.length === 0) return
   395→    if (!body) return
   396→    if (!channelId) return
   397→    try {
   398→      if (editingMessage) {
   399→        await editMessage(editingMessage.id, body)
   400→        toast.success('Message updated')
   401→        return
   402→      }
   403→      await sendMessage(body, files.length > 0 ? files.map((f) => f.id) : undefined, parentId)
   404→      setText('')
   405→      setDraft(draftKey, '')
   406→      setFiles([])
   407→      requestAnimationFrame(() => {
   408→        const el = textareaRef.current
   409→        if (el) {
   410→          el.style.height = 'auto'
   411→          el.focus()
   412→        }
   413→      })
   414→    } catch (err) {
   415→      toast.error(err instanceof Error ? err.message : 'Could not send message')
   416→    }
   417→  }
   418→
   419→  const handleKeyDown = (event: React.KeyboardEvent<HTMLTextAreaElement>) => {
   420→    if (menu) {
   421→      if (event.key === 'ArrowDown') {
   422→        event.preventDefault()
   423→        setMenuIndex((i) => i + 1)
   424→        return
   425→      }
   426→      if (event.key === 'ArrowUp') {
   427→        event.preventDefault()
   428→        setMenuIndex((i) => Math.max(0, i - 1))
   429→        return
   430→      }
   431→      if (event.key === 'Enter' || event.key === 'Tab') {
   432→        event.preventDefault()
   433→        if (menu.kind === 'mention') {
   434→          const q = menu.query.toLowerCase()
   435→          const filtered = users.filter(
   436→            (user) => !q || user.name.toLowerCase().includes(q) || (user.handle ?? '').includes(q),
   437→          )
   438→          const offset = q.length >= 2 ? 2 : 0
   439→          if (menuIndex === 0 && q.length >= 2) selectMention('channel')
   440→          else if (menuIndex === 1 && q.length >= 2) selectMention('here')
   441→          else if (filtered[menuIndex - offset]) selectMention(filtered[menuIndex - offset])
   442→        } else {
   443→          const q = menu.query.toLowerCase()
   444→          const match = channels.find(
   445→            (c) => c.kind !== 'dm' && c.kind !== 'group_dm' && (c.slug.includes(q) || c.name.toLowerCase().includes(q)),
   446→          )
   447→          if (match) selectChannel(match.slug)
   448→        }
   449→        return
   450→      }
   451→      if (event.key === 'Escape') {
   452→        event.preventDefault()
   453→        setMenu(null)
   454→        return
   455→      }
   456→    }
   457→
   458→    if (event.key === 'Enter' && !event.shiftKey) {
   459→      event.preventDefault()
   460→      void submit()
   461→      return
   462→    }
   463→    if (event.key === 'Escape' && editingMessage) {
   464→      event.preventDefault()
   465→      setEditingMessageId(null)
   466→      setText(drafts[draftKey] ?? '')
   467→      return
   468→    }
   469→    if (event.key === 'ArrowUp' && !text && !editingMessage && !parentId) {
   470→      // edit my own last message in this channel
   471→      const mine = [...messages].reverse().find((m) => m.sender?.id === me?.id && !m.deletedAt)
   472→      if (mine) {
   473→        event.preventDefault()
   474→        setEditingMessageId(mine.id)
   475→      }
   476→    }
   477→  }
   478→
   479→  if (!channel || channel.isArchived) {
   480→    return (
   481→      <div className="shrink-0 px-4 pb-4 md:px-6">
   482→        <div className="rounded-xl border border-dashed border-border bg-muted/30 px-4 py-3 text-center text-sm text-muted-foreground">
   483→          This channel is archived — history is read-only.
   484→        </div>
   485→      </div>
   486→    )
   487→  }
   488→
   489→  const displayPlaceholder =
   490→    placeholder ??
   491→    (channel.kind === 'dm' || channel.kind === 'group_dm'
   492→      ? channel.members?.length === 1
   493→        ? `Message ${channel.members[0].name}${channel.members[0].kind === 'agent' ? ' — try asking anything' : ''}`
   494→        : 'Message group'
   495→      : `Message #${channel.name}`)
   496→
   497→  const toolbarButton =
   498→    'flex h-7 w-7 items-center justify-center rounded-md text-muted-foreground transition-colors duration-150 hover:bg-accent hover:text-foreground'
   499→
   500→  return (
   501→    <div className="relative shrink-0 px-4 pb-4 md:px-6">
   502→      {!parentId && channelId && <TypingIndicator channelId={channelId} />}
   503→
   504→      {/* edit bar */}
   505→      {editingMessage && (
   506→        <div className="mb-1.5 flex items-center gap-2 rounded-t-xl border border-b-0 border-amber-500/40 bg-amber-500/10 px-3 py-1.5 text-xs font-medium text-amber-700 dark:text-amber-300">
   507→          <span className="flex-1">
   508→            Editing message
   509→            <span className="ml-1.5 font-normal text-amber-600/70 dark:text-amber-400/70">Esc to cancel</span>
   510→          </span>
   511→          <button
   512→            type="button"
   513→            aria-label="Cancel edit"
   514→            onClick={() => {
   515→              setEditingMessageId(null)
   516→              setText(drafts[draftKey] ?? '')
   517→            }}
   518→            className="flex h-6 w-6 items-center justify-center rounded hover:bg-amber-500/20"
   519→          >
   520→            <X className="h-3.5 w-3.5" aria-hidden />
   521→          </button>
   522→          <button
   523→            type="button"
   524→            aria-label="Save edit"
   525→            onClick={() => void submit()}
   526→            className="flex h-6 w-6 items-center justify-center rounded hover:bg-amber-500/20"
   527→          >
   528→            <Check className="h-3.5 w-3.5" aria-hidden />
   529→          </button>
   530→        </div>
   531→      )}
   532→
   533→      <div
   534→        className={cn(
   535→          'relative rounded-xl border bg-card shadow-sm transition-colors duration-150 focus-within:border-emerald-500/50',
   536→          editingMessage ? 'rounded-t-none border-amber-500/40' : 'border-border',
   537→          dragging && 'border-emerald-500 ring-2 ring-emerald-500/30',
   538→        )}
   539→        onDragOver={(event) => {
   540→          event.preventDefault()
   541→          setDragging(true)
   542→        }}
   543→        onDragLeave={() => setDragging(false)}
   544→        onDrop={(event) => {
   545→          event.preventDefault()
   546→          setDragging(false)
   547→          if (event.dataTransfer.files.length > 0) void uploadFiles(event.dataTransfer.files)
   548→        }}
   549→      >
   550→        {/* drop overlay */}
   551→        {dragging && (
   552→          <div className="absolute inset-0 z-20 flex flex-col items-center justify-center gap-1 rounded-xl border-2 border-dashed border-emerald-500 bg-emerald-500/10 backdrop-blur-sm">

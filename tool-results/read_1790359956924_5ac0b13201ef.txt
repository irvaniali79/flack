     1→// ─── Connector catalog (Slack-style App Directory) ───────────────────────────
     2→// A connector is an external service (Google Calendar, GitHub, …) installed
     3→// into the workspace. Each install ("connection") links an account label to a
     4→// destination channel and posts its events there as a bot "app" user
     5→// (User.kind = 'app'), exactly like Slack app messages.
     6→//
     7→// This module is server-safe (no React). Icons are referenced by key — the
     8→// client maps them to lucide components in the Connectors view.
     9→import { db } from '@/lib/db'
    10→import { emitToChannel } from '@/lib/realtime-server'
    11→import { serializeMessage } from '@/lib/serialize'
    12→import type { ConnectorConnection, User } from '@prisma/client'
    13→
    14→// ─── Types ───────────────────────────────────────────────────────────────────
    15→
    16→export interface ConnectorEventDef {
    17→  id: string
    18→  label: string
    19→  description: string
    20→  defaultOn: boolean
    21→}
    22→
    23→export interface ConnectorDef {
    24→  id: string
    25→  name: string
    26→  tagline: string
    27→  description: string
    28→  category: string
    29→  /** icon key resolved by the client icon map */
    30→  icon: string
    31→  /** brand hex used for the app tile */
    32→  color: string
    33→  /** demo OAuth scope labels shown in the connect flow */
    34→  scopes: string[]
    35→  /** realistic demo account labels for the connect flow */
    36→  sampleAccounts: string[]
    37→  events: ConnectorEventDef[]
    38→}
    39→
    40→/** A concrete event instance — the content of one app message */
    41→export interface ConnectorEventInstance {
    42→  eventId: string
    43→  title: string
    44→  /** plain-text version (searchable, notifications, sidebar previews) */
    45→  body: string
    46→  fields?: { label: string; value: string }[]
    47→  actions?: { label: string; style: 'primary' | 'default' }[]
    48→  footer?: string
    49→}

   883→/** Pick a realistic sample event (matching eventId when provided). */
   884→export function pickEventInstance(def: ConnectorDef, eventId?: string): ConnectorEventInstance | null {
   885→  const samples = CONNECTOR_SAMPLES[def.id] ?? []
   886→  if (samples.length === 0) return null
   887→  const pool = eventId ? samples.filter((s) => s.eventId === eventId) : samples
   888→  if (pool.length === 0) return null
   889→  return pool[Math.floor(Math.random() * pool.length)]
   890→}

   922→/**
   923→ * Post one connector event into the connection's channel as the app user,
   924→ * with realtime fan-out. Shared by the simulator route and the connect flow.
   925→ */
   926→export async function postConnectorEvent(
   927→  connection: ConnectorConnection,
   928→  instance: ConnectorEventInstance,
   929→): Promise<{ message: ReturnType<typeof serializeMessage>; channelName: string }> {
   930→  const def = getConnectorDef(connection.connectorId)
   931→  const payload = {
   932→    event: instance.eventId,
   933→    title: instance.title,
   934→    fields: instance.fields ?? [],
   935→    actions: instance.actions ?? [],
   936→    footer: instance.footer ?? '',
   937→  }
   938→  const message = await db.message.create({
   939→    data: {
   940→      channelId: connection.channelId,
   941→      senderId: connection.appUserId,
   942→      body: instance.body,
   943→      mentions: JSON.stringify({ userIds: [], specials: [] }),
   944→      connectorId: connection.connectorId,
   945→      connectorPayload: JSON.stringify(payload),
   946→    },
   947→    include: messageInclude,
   948→  })
   949→  const dto = serializeMessage(
   950→    message as unknown as Parameters<typeof serializeMessage>[0],
   951→  )
   952→  void emitToChannel(connection.channelId, 'message:new', { message: dto }).catch(() => {})
   953→  return { message: dto, channelName: def?.name ?? connection.connectorId }
   954→}

   964→export function parseEventSubs(raw: string | null): Record<string, boolean> {
   965→  if (!raw) return {}
   966→  try {
   967→    const parsed = JSON.parse(raw) as Record<string, unknown>
   968→    const out: Record<string, boolean> = {}
   969→    for (const [k, v] of Object.entries(parsed)) out[k] = v === true
   970→    return out
   971→  } catch {
   972→    return {}
   973→  }
   974→}
